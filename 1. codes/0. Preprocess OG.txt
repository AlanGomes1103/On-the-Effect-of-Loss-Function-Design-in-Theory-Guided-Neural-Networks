def parse_filename(fname):
    """
    Parse Mach number and angle/parameter from filename.
    Works for oblique (wedge angle) and bow shocks (nose radius or just Mach).
    Returns (mach, angle_or_param) — angle_or_param is 0.0 for bow shocks
    with no geometric parameter in the filename.
    """
    fname = os.path.basename(fname)

    # Pattern 1: wedge_M2_0_angle12_5.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_(\d+)_angle(\d+)_(\d+)', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), float(f"{m.group(3)}.{m.group(4)}")

    # Pattern 1b: wedge_M2_0_angle7.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_(\d+)_angle(\d+)(?:[^\d_]|$)', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), float(m.group(3))

    # Pattern 2: wedge_M2.5_angle10.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+\.\d+)_angle(\d+(?:\.\d+)?)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), float(m.group(2))

    # Pattern 3: wedge_M3_angle10.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_angle(\d+(?:\.\d+)?)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), float(m.group(2))

    # Pattern 4: bow_M2_5.xlsx or bow_M3.xlsx (no angle parameter)
    m = re.search(r'(?:bow|blunt|cylinder|sphere)_M(\d+)_(\d+)', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), 0.0

    m = re.search(r'(?:bow|blunt|cylinder|sphere)_M(\d+\.\d+)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), 0.0

    m = re.search(r'(?:bow|blunt|cylinder|sphere)_M(\d+)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), 0.0

    # Pattern 5: generic M(\d) anywhere — last resort
    m = re.search(r'_M(\d+)_(\d+)_', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), 0.0

    return None, None
    
def compute_theoretical_shock_angle(mach, wedge_angle_deg):
    """
    Solve θ-β-M for oblique shocks. Returns NaN for bow shocks (angle=0)
    or if detached/no solution.
    """
    if wedge_angle_deg <= 0:
        return np.nan   # bow shock — no single β
    gamma = 1.4
    try:
        mach_angle = np.degrees(np.arcsin(1.0 / mach))
    except (ValueError, ZeroDivisionError):
        return np.nan

    def shock_relation(beta_deg):
        beta_rad  = np.radians(beta_deg)
        wedge_rad = np.radians(wedge_angle_deg)
        tan_theta = (2 / np.tan(beta_rad) *
                     (mach**2 * np.sin(beta_rad)**2 - 1) /
                     (mach**2 * (gamma + np.cos(2 * beta_rad)) + 2))
        return np.tan(wedge_rad) - tan_theta

    bv = np.linspace(mach_angle + 0.5, 89, 500)
    sv = [shock_relation(b) for b in bv]
    sc = [(bv[i], bv[i+1]) for i in range(len(sv)-1) if sv[i]*sv[i+1] < 0]
    if not sc:
        return np.nan
    try:
        return brentq(shock_relation, sc[0][0], sc[0][1])
    except Exception:
        return np.nan

def detect_shock_type(fname):
    """
    Detect shock type from filename.
    Returns 'oblique', 'bow', or 'unknown'.
    Filename conventions:
      oblique: contains 'wedge' or 'oblique'
      bow:     contains 'bow' or 'cylinder' or 'blunt'
    """
    fname_lower = os.path.basename(fname).lower()
    if any(k in fname_lower for k in ['wedge', 'oblique']):
        return 'oblique'
    if any(k in fname_lower for k in ['bow', 'cylinder', 'blunt', 'sphere']):
        return 'bow'
    return 'unknown'

def extract_line_profiles(df, x_positions, y_positions):
    """
    Extract pressure profiles along specific x and y lines for validation.
    Returns dictionaries with profile data.
    """
    profiles = {
        'x_lines': {},
        'y_lines': {}
    }
    
    # Extract along constant x lines
    for x in x_positions:
        # Get points within tolerance of this x
        tolerance = 0.002  # 2mm tolerance
        mask = np.abs(df['x-coordinate'] - x) < tolerance
        if mask.any():
            profile_df = df[mask].sort_values('y-coordinate')
            profiles['x_lines'][x] = {
                'y': profile_df['y-coordinate'].values,
                'pressure': profile_df['pressure'].values,
                'velocity': profile_df['velocity-magnitude'].values,
                'temperature': profile_df['temperature'].values,
                'density': profile_df['density'].values
            }
    
    # Extract along constant y lines
    for y in y_positions:
        tolerance = 0.002
        mask = np.abs(df['y-coordinate'] - y) < tolerance
        if mask.any():
            profile_df = df[mask].sort_values('x-coordinate')
            profiles['y_lines'][y] = {
                'x': profile_df['x-coordinate'].values,
                'pressure': profile_df['pressure'].values,
                'velocity': profile_df['velocity-magnitude'].values,
                'temperature': profile_df['temperature'].values,
                'density': profile_df['density'].values
            }
    
    return profiles

def compute_gradients(df, coords_cols=['x-coordinate', 'y-coordinate']):
    """
    Compute combined shock-detection gradient using both pressure and density.
    Normalises each field so neither dominates, then takes the maximum.
    This gives a much stronger and cleaner shock signal than pressure alone.
    """
    coords = df[coords_cols].values
    tree = cKDTree(coords)
    k = min(8, len(df))  # more neighbours = smoother, less noise-sensitive
    distances, indices = tree.query(coords, k=k)

    def _field_gradient(values):
        val_range = values.max() - values.min() + 1e-10
        norm_values = values / val_range          # normalise to [0,1] range
        grads = np.zeros(len(df))
        for i in range(len(df)):
            nbr_vals = norm_values[indices[i][1:]]
            wts = 1.0 / (distances[i][1:] + 1e-10)
            grads[i] = np.sum(np.abs(nbr_vals - norm_values[i]) * wts) / (np.sum(wts) + 1e-10)
        return grads

    grad_pressure    = _field_gradient(df['pressure'].values)
    grad_density     = _field_gradient(df['density'].values)
    grad_temperature = _field_gradient(df['temperature'].values)

    # Combined: take element-wise maximum so ANY field discontinuity is caught
    combined = np.maximum(grad_pressure, np.maximum(grad_density, grad_temperature))
    return combined

def remove_region_outliers(df, z_thresh=3.0):
    """
    Within each region (pre/post/transition), remove physically implausible points
    that are statistical outliers across all flow variables.
    
    Uses z-score per region per variable. Points that are outliers in ANY
    variable are flagged and removed. This catches:
      - Mesh boundary artefacts
      - Solver convergence glitches  
      - Mis-classified points near the shock that slipped through
    
    Pre-shock outlier detection is looser (z_thresh * 1.5) because freestream
    is genuinely uniform — any deviation there is almost certainly an artefact.
    """
    FLOW_VARS = ['pressure', 'density', 'temperature', 'velocity-magnitude', 'mach-number']
    
    clean_parts = []
    removal_stats = {}
    
    for region in ['pre', 'post', 'transition']:
        region_df = df[df['region'] == region].copy()
        if len(region_df) == 0:
            continue
        
        # Pre-shock is spatially uniform → tighter outlier rejection
        thresh = z_thresh * 1.5 if region == 'pre' else z_thresh
        
        outlier_mask = pd.Series(False, index=region_df.index)
        
        for var in FLOW_VARS:
            if var not in region_df.columns:
                continue
            vals = region_df[var]
            mean, std = vals.mean(), vals.std()
            if std < 1e-10:   # perfectly uniform → nothing to flag
                continue
            z_scores = np.abs((vals - mean) / std)
            outlier_mask |= (z_scores > thresh)
        
        n_before = len(region_df)
        clean_region = region_df[~outlier_mask]
        n_removed = n_before - len(clean_region)
        
        removal_stats[region] = {
            'before': n_before,
            'removed': n_removed,
            'fraction_removed': n_removed / n_before if n_before > 0 else 0
        }
        
        clean_parts.append(clean_region)
    
    clean_df = pd.concat(clean_parts, ignore_index=False)  # keep original index
    return clean_df, removal_stats

def validate_rankine_hugoniot(df, mach, wedge_angle_deg, shock_angle_deg=None):
    """
    Validate region classification using Rankine-Hugoniot jump conditions.
    This is the physics ground truth — if P2/P1 and T2/T1 match theory,
    the pre/post regions are correctly identified.

    Returns a dict with ratios, theoretical values, and pass/fail flags.
    """
    gamma = 1.4

    pre  = df[df['region'] == 'pre']
    post = df[df['region'] == 'post']

    if len(pre) == 0 or len(post) == 0:
        return {'valid': False, 'reason': 'missing region'}

    p1 = pre['pressure'].mean()
    T1 = pre['temperature'].mean()
    rho1 = pre['density'].mean()
    p2 = post['pressure'].mean()
    T2 = post['temperature'].mean()
    rho2 = post['density'].mean()

   # Use provided shock angle or fall back to theoretical

    # Bow shocks have no single wedge angle — skip theoretical β derivation
    if wedge_angle_deg <= 0:
        return {'valid': False, 'reason': 'bow shock — no single beta'}
    
    if shock_angle_deg is None or (isinstance(shock_angle_deg, float) and np.isnan(shock_angle_deg)):
        shock_angle_deg = compute_theoretical_shock_angle(mach, wedge_angle_deg)

    # If theoretical also failed (brentq no bracket), derive shock angle
    # directly from the data pressure ratio using R-H inverse
    if shock_angle_deg is None or (isinstance(shock_angle_deg, float) and np.isnan(shock_angle_deg)):
        # Invert P2/P1 = (2γMn1² - (γ-1))/(γ+1) to get Mn1, then β = arcsin(Mn1/M)
        p_ratio_measured = p2 / p1
        Mn1_sq = (p_ratio_measured * (gamma + 1) + (gamma - 1)) / (2 * gamma)
        if Mn1_sq > 0:
            Mn1_derived = np.sqrt(Mn1_sq)
            shock_angle_deg = np.degrees(np.arcsin(np.clip(Mn1_derived / mach, 0, 1)))
        else:
            return {'valid': False, 'reason': 'cannot determine shock angle'}

    beta_rad = np.radians(shock_angle_deg)
    Mn1 = mach * np.sin(beta_rad)

    # Rankine-Hugoniot relations
    p_ratio_theory   = (2*gamma*Mn1**2 - (gamma-1)) / (gamma+1)
    T_ratio_theory   = ((2*gamma*Mn1**2-(gamma-1)) * ((gamma-1)*Mn1**2+2)) / ((gamma+1)**2 * Mn1**2)
    rho_ratio_theory = (gamma+1)*Mn1**2 / ((gamma-1)*Mn1**2 + 2)

    p_ratio_data   = p2 / p1
    T_ratio_data   = T2 / T1
    rho_ratio_data = rho2 / rho1

    # Tolerance: 5% is generous for a CFD mesh average
    tol = 0.05
    p_ok   = abs(p_ratio_data   - p_ratio_theory)   / p_ratio_theory   < tol
    T_ok   = abs(T_ratio_data   - T_ratio_theory)   / T_ratio_theory   < tol
    rho_ok = abs(rho_ratio_data - rho_ratio_theory) / rho_ratio_theory < tol

    return {
        'valid': p_ok and T_ok and rho_ok,
        'shock_angle_used': shock_angle_deg,
        'Mn1': Mn1,
        'p_ratio_data':   p_ratio_data,   'p_ratio_theory':   p_ratio_theory,   'p_ok':   p_ok,
        'T_ratio_data':   T_ratio_data,   'T_ratio_theory':   T_ratio_theory,   'T_ok':   T_ok,
        'rho_ratio_data': rho_ratio_data, 'rho_ratio_theory': rho_ratio_theory, 'rho_ok': rho_ok,
        'p_error_pct':   abs(p_ratio_data-p_ratio_theory)/p_ratio_theory*100,
        'T_error_pct':   abs(T_ratio_data-T_ratio_theory)/T_ratio_theory*100,
        'rho_error_pct': abs(rho_ratio_data-rho_ratio_theory)/rho_ratio_theory*100,
    }

def fit_shock_line(points, slope_only=True):
    """Fit a line through shock candidate points."""
    if len(points) < 2:
        return None, None, None
    
    X = points[:, 0].reshape(-1, 1)
    y = points[:, 1]
    
    ransac = RANSACRegressor(random_state=RANDOM_SEED, residual_threshold=0.01)
    ransac.fit(X, y)
    
    slope = ransac.estimator_.coef_[0]
    intercept = ransac.estimator_.intercept_
    
    if slope_only:
        # Shock must pass through wedge apex at origin
        slope = np.sum(X.flatten() * y) / np.sum(X.flatten()**2)
        intercept = 0.0
    
    angle_deg = np.degrees(np.arctan(slope))
    
    return slope, intercept, angle_deg

def verify_region_separation(df, classifier_version):
    """
    Verify that regions are properly separated using multiple metrics.
    Returns a verification report dictionary.
    """
    verification = {
        'version': classifier_version,
        'tests': {}
    }
    
    # Test 1: Check if pre-shock and post-shock pressure distributions are distinct
    pre_pressure = df[df['region'] == 'pre']['pressure'].values
    post_pressure = df[df['region'] == 'post']['pressure'].values
    
    if len(pre_pressure) > 0 and len(post_pressure) > 0:
        # Kolmogorov-Smirnov test for distribution difference
        from scipy.stats import ks_2samp
        ks_stat, ks_pvalue = ks_2samp(pre_pressure, post_pressure)
        
        verification['tests']['pressure_distribution'] = {
            'ks_statistic': ks_stat,
            'ks_pvalue': ks_pvalue,
            'pre_mean': pre_pressure.mean(),
            'post_mean': post_pressure.mean(),
            'pre_std': pre_pressure.std(),
            'post_std': post_pressure.std(),
            'distinct': ks_stat > 0.3 and ks_pvalue < 0.05
        }
    
 # Test 2: Check separation along shock-normal direction
    # For an oblique shock through the origin, the shock-normal direction is (-sin β, cos β)
    # where β is estimated from the mean pressure ratio or from the data directly.
    # We approximate β using the centroid positions of pre and post regions.
    pre_df  = df[df['region'] == 'pre']
    post_df = df[df['region'] == 'post']

    if len(pre_df) > 0 and len(post_df) > 0:
        # Estimate shock slope from centroid vector between region centroids
        pre_cx,  pre_cy  = pre_df['x-coordinate'].mean(),  pre_df['y-coordinate'].mean()
        post_cx, post_cy = post_df['x-coordinate'].mean(), post_df['y-coordinate'].mean()

        # Project all points onto the shock-normal axis (perpendicular to centroid vector)
        dx, dy = post_cx - pre_cx, post_cy - pre_cy
        norm = np.sqrt(dx**2 + dy**2) + 1e-10
        nx, ny = dx / norm, dy / norm          # unit normal pointing pre→post

        pre_proj  = pre_df['x-coordinate'].values  * nx + pre_df['y-coordinate'].values  * ny
        post_proj = post_df['x-coordinate'].values * nx + post_df['y-coordinate'].values * ny

        pre_proj_max  = pre_proj.max()
        post_proj_min = post_proj.min()
        separation_gap = post_proj_min - pre_proj_max  # positive = clean separation

        verification['tests']['spatial_separation'] = {
            'pre_proj_range':  (float(pre_proj.min()),  float(pre_proj_max)),
            'post_proj_range': (float(post_proj_min), float(post_proj.max())),
            'separated': separation_gap > 0,
            'gap': float(separation_gap)
        }
    else:
        verification['tests']['spatial_separation'] = {
            'pre_proj_range': (0, 0), 'post_proj_range': (0, 0),
            'separated': False, 'gap': 0.0
        }
    
    # Test 3: Check transition region continuity
    trans_points = df[df['region'] == 'transition']
    if len(trans_points) > 0:
        trans_y_range = (trans_points['y-coordinate'].min(), trans_points['y-coordinate'].max())
        verification['tests']['transition_region'] = {
            'num_points': len(trans_points),
            'y_range': trans_y_range,
            'width': trans_points['y-coordinate'].max() - trans_points['y-coordinate'].min()
        }
    
    # Test 4: Check if points follow expected geometry (within bounds)
    y_max_check = GEOMETRY['y_max'] if GEOMETRY['y_max'] is not None else df['y-coordinate'].max()
    within_bounds = ((df['x-coordinate'] >= GEOMETRY['x_min']) &
                     (df['x-coordinate'] <= GEOMETRY['x_max']) &
                     (df['y-coordinate'] >= GEOMETRY['y_min']) &
                     (df['y-coordinate'] <= y_max_check))
    
    verification['tests']['geometry_bounds'] = {
        'points_in_bounds': within_bounds.sum(),
        'total_points': len(df),
        'fraction_in_bounds': within_bounds.sum() / len(df),
        'x_range': (df['x-coordinate'].min(), df['x-coordinate'].max()),
        'y_range': (df['y-coordinate'].min(), df['y-coordinate'].max())
    }
    
    return verification

def analyze_parameter_relationship(classifier_results):
    """
    Analyze how changes in wedge angle and Mach number affect shock behavior.
    Returns analysis dictionary for visualization.
    """
    analysis = {
        'by_mach': {},
        'by_angle': {},
        'correlations': {}
    }
    
    # Group by Mach number
    for mach in sorted(classifier_results['mach'].unique()):
        mach_data = classifier_results[classifier_results['mach'] == mach]
        
        # Collect shock angles for this Mach number
        shock_angles = mach_data['shock_angle'].dropna().values
        
        if len(shock_angles) > 0:
            analysis['by_mach'][mach] = {
                'mean_angle': shock_angles.mean(),
                'std_angle': shock_angles.std(),
                'num_samples': len(shock_angles),
                'angle_range': (shock_angles.min(), shock_angles.max()),
                'wedge_angles': mach_data['wedge_angle'].unique()
            }
    
    # Group by wedge angle
    for angle in sorted(classifier_results['wedge_angle'].unique()):
        angle_data = classifier_results[classifier_results['wedge_angle'] == angle]
        
        shock_angles = angle_data['shock_angle'].dropna().values
        
        if len(shock_angles) > 0:
            analysis['by_angle'][angle] = {
                'mean_angle': shock_angles.mean(),
                'std_angle': shock_angles.std(),
                'num_samples': len(shock_angles),
                'mach_numbers': angle_data['mach'].unique()
            }
    
    # Calculate correlations if enough data
    if len(classifier_results) > 10:
        # Correlation between wedge angle and shock angle
        valid_data = classifier_results.dropna(subset=['shock_angle', 'wedge_angle'])
        if len(valid_data) > 5:
            from scipy.stats import pearsonr
            corr_angle, p_angle = pearsonr(valid_data['wedge_angle'], valid_data['shock_angle'])
            analysis['correlations']['wedge_vs_shock'] = {
                'correlation': corr_angle,
                'p_value': p_angle,
                'n_points': len(valid_data)
            }
        
        # Correlation between Mach number and shock angle
        valid_data_mach = classifier_results.dropna(subset=['shock_angle', 'mach'])
        if len(valid_data_mach) > 5:
            corr_mach, p_mach = pearsonr(valid_data_mach['mach'], valid_data_mach['shock_angle'])
            analysis['correlations']['mach_vs_shock'] = {
                'correlation': corr_mach,
                'p_value': p_mach,
                'n_points': len(valid_data_mach)
            }
    
    return analysis

class ShockClassifier:
    """Base class for shock classification strategies."""
    
    def __init__(self, name, version_id):
        self.name = name
        self.version_id = version_id
        self.diagnostics = {}
    
    def classify(self, df):
        raise NotImplementedError
    
    def sample_points(self, df):
        raise NotImplementedError
    
    def get_diagnostics(self):
        return self.diagnostics

class Version1_PressureBased(ShockClassifier):
    """Version 1: Original pressure-based classification."""
    
    def __init__(self):
        super().__init__("Pressure-Based", 1)
        self.pre_threshold = None
        self.post_threshold = None
    
    def classify(self, df):
        p = df['pressure'].values
        p_min, p_max = p.min(), p.max()
        jump = p_max - p_min
        p_mid = p_min + 0.5 * jump
        band = 0.10 * jump
        
        pre_threshold = p_mid - band
        post_threshold = p_mid + band
        
        labels = np.where(p <= pre_threshold, 'pre',
                 np.where(p >= post_threshold, 'post', 'transition'))
        
        self.pre_threshold = pre_threshold
        self.post_threshold = post_threshold
        self.diagnostics = {
            'p_min': p_min, 'p_max': p_max,
            'p_mid': p_mid, 'band_width': band,
            'pre_threshold': pre_threshold,
            'post_threshold': post_threshold
        }
        
        return pd.Series(labels, index=df.index, name='region')
    
    def sample_points(self, df):
        pre_df = df[df['region'] == 'pre']
        post_df = df[df['region'] == 'post']
        trans_df = df[df['region'] == 'transition']
        
        pre_sample = pre_df.sample(frac=PRE_SHOCK_SAMPLE_FRAC, random_state=RANDOM_SEED)
        trans_sample = trans_df
        
        if len(post_df) > 0:
            x_edges = np.linspace(post_df['x-coordinate'].min(), post_df['x-coordinate'].max(), POST_SHOCK_SAMPLE_GRID)
            y_edges = np.linspace(post_df['y-coordinate'].min(), post_df['y-coordinate'].max(), POST_SHOCK_SAMPLE_GRID)
            
            post_sample_indices = []
            for i in range(len(x_edges)-1):
                for j in range(len(y_edges)-1):
                    mask = ((post_df['x-coordinate'] >= x_edges[i]) & 
                           (post_df['x-coordinate'] < x_edges[i+1]) &
                           (post_df['y-coordinate'] >= y_edges[j]) & 
                           (post_df['y-coordinate'] < y_edges[j+1]))
                    cell_points = post_df[mask]
                    if len(cell_points) > 0:
                        post_sample_indices.append(cell_points.sample(1).index[0])
            
            post_sample = post_df.loc[post_sample_indices] if post_sample_indices else post_df.sample(
                n=min(500, len(post_df)), random_state=RANDOM_SEED)
            # Guarantee a minimum — sparse grids may yield very few cells
            if len(post_sample) < 200 and len(post_df) >= 200:
                post_sample = post_df.sample(n=200, random_state=RANDOM_SEED)
        else:
            post_sample = post_df
        
        sampled = pd.concat([pre_sample, post_sample, trans_sample], ignore_index=True)
        
        self.diagnostics.update({
            'pre_original': len(pre_df), 'pre_sampled': len(pre_sample),
            'post_original': len(post_df), 'post_sampled': len(post_sample),
            'transition_original': len(trans_df), 'transition_sampled': len(trans_sample)
        })
        
        return sampled

class Version2_GradientBased(ShockClassifier):
    """Version 2: Gradient-based shock detection."""
    
    def __init__(self):
        super().__init__("Gradient-Based", 2)
        self.gradient_threshold = None
        self.shock_line_slope = None
        self.shock_angle = None
    
    def classify(self, df):
        gradients = compute_gradients(df)
        df = df.copy()
        df['gradient'] = gradients
        
        threshold = np.percentile(gradients, GRADIENT_THRESHOLD_PERCENTILE)
        candidate_mask = gradients > threshold
        
        if candidate_mask.sum() > 5:
            candidate_coords = df[candidate_mask][['x-coordinate', 'y-coordinate']].values
            clustering = DBSCAN(eps=SHOCK_CLUSTER_EPS, min_samples=3).fit(candidate_coords)
            
            labels_cluster = clustering.labels_
            unique_clusters = np.unique(labels_cluster[labels_cluster >= 0])
            if len(unique_clusters) > 0:
                # Select cluster whose points best fit a line through the origin
                # (shock must pass through wedge apex). Score = R² of origin-forced fit.
                best_cluster = None
                best_r2 = -np.inf
                for cid in unique_clusters:
                    cmask = labels_cluster == cid
                    cpts = candidate_coords[cmask]
                    if len(cpts) < 3:
                        continue
                    # Origin-forced slope
                    cx, cy = cpts[:, 0], cpts[:, 1]
                    s = np.sum(cx * cy) / (np.sum(cx ** 2) + 1e-10)
                    residuals = cy - s * cx
                    ss_res = np.sum(residuals ** 2)
                    ss_tot = np.sum((cy - cy.mean()) ** 2) + 1e-10
                    r2 = 1 - ss_res / ss_tot
                    if r2 > best_r2:
                        best_r2 = r2
                        best_cluster = cid

                shock_mask = labels_cluster == best_cluster
                shock_points = candidate_coords[shock_mask]
                slope, intercept, angle = fit_shock_line(shock_points)
                self.shock_line_slope = slope
                self.shock_angle = angle
                
                if slope is not None:
                    y_shock = slope * df['x-coordinate'].values
                    y_point = df['y-coordinate'].values

                    # Below shock line = near wedge surface = post-shock (compressed)
                    # Above shock line = free stream = pre-shock
                    labels = np.where(y_point < y_shock - TRANSITION_BAND_WIDTH/2, 'post',
                             np.where(y_point > y_shock + TRANSITION_BAND_WIDTH/2, 'pre', 'transition'))

                    self.diagnostics['shock_angle'] = angle
                else:
                    return Version1_PressureBased().classify(df)
            else:
                return Version1_PressureBased().classify(df)
        else:
            return Version1_PressureBased().classify(df)
        
        self.gradient_threshold = threshold
        self.diagnostics['gradient_threshold'] = threshold
        self.diagnostics['candidate_points'] = candidate_mask.sum()
        
        return pd.Series(labels, index=df.index, name='region')
    
    def sample_points(self, df):
        v1 = Version1_PressureBased()
        return v1.sample_points(df)

class Version3_ShockLineFitting(ShockClassifier):
    """Version 3: Shock line fitting with region classification."""
    
    def __init__(self):
        super().__init__("Shock Line Fitting", 3)
        self.shock_slope = None
        self.shock_intercept = None
        self.shock_angle = None
    
    def classify(self, df):
        v1 = Version1_PressureBased()
        regions = v1.classify(df)
        df = df.copy()
        df['region_temp'] = regions
        
        candidate_df = df[df['region_temp'] == 'transition']
        
        if len(candidate_df) > 10:
            candidate_coords = candidate_df[['x-coordinate', 'y-coordinate']].values
            slope, intercept, angle = fit_shock_line(candidate_coords, slope_only=True)
            
            if slope is not None:
                self.shock_slope = slope
                self.shock_intercept = intercept
                self.shock_angle = angle

                y_shock = slope * df['x-coordinate'].values + intercept
                y_point = df['y-coordinate'].values

                # Below shock line = near wedge surface = post-shock (compressed)
                # Above shock line = free stream = pre-shock
                labels = np.where(y_point < y_shock - TRANSITION_BAND_WIDTH/2, 'post',
                         np.where(y_point > y_shock + TRANSITION_BAND_WIDTH/2, 'pre', 'transition'))
            else:
                return Version1_PressureBased().classify(df)
        else:
            return Version1_PressureBased().classify(df)
        
        self.diagnostics = {
            'shock_slope': slope,
            'shock_intercept': intercept,
            'shock_angle': angle,
            'candidate_points': len(candidate_df)
        }
        
        return pd.Series(labels, index=df.index, name='region')
    
    def sample_points(self, df):
        v1 = Version1_PressureBased()
        return v1.sample_points(df)

class Version4_Hybrid(ShockClassifier):
    """Version 4: Hybrid approach combining pressure and geometry."""
    
    def __init__(self):
        super().__init__("Hybrid", 4)
        self.pressure_jump = None
        self.shock_angle = None
    
    def classify(self, df):
        v3 = Version3_ShockLineFitting()
        regions_geom = v3.classify(df)
        
        v1 = Version1_PressureBased()
        regions_pressure = v1.classify(df)
        
        df = df.copy()
        df['region_geom'] = regions_geom
        df['region_pressure'] = regions_pressure
        
        consistent_mask = df['region_geom'] == df['region_pressure']
        
        if not consistent_mask.all():
            gradients = compute_gradients(df)
            high_grad_mask = gradients > np.percentile(gradients, 95)
            
            labels = df['region_geom'].copy()
            labels[~consistent_mask & high_grad_mask] = 'transition'
            labels[~consistent_mask & ~high_grad_mask] = df['region_pressure'][~consistent_mask]
        else:
            labels = df['region_geom']
        
        self.shock_angle = v3.shock_angle
        self.pressure_jump = v1.diagnostics.get('p_max', 0) - v1.diagnostics.get('p_min', 0)
        
        self.diagnostics = {
            'shock_angle': self.shock_angle,
            'pressure_jump': self.pressure_jump,
            'consistency_rate': consistent_mask.mean()
        }
        
        return labels
    
    def sample_points(self, df):
        v1 = Version1_PressureBased()
        sampled = v1.sample_points(df)
        
        post_df = df[df['region'] == 'post']
        if len(post_df) > 0 and self.shock_angle is not None:
            slope = np.tan(np.radians(self.shock_angle))
            y_shock = slope * post_df['x-coordinate'].values
            dist_to_shock = np.abs(post_df['y-coordinate'].values - y_shock) / np.sqrt(1 + slope**2)
            
            near_shock = post_df[dist_to_shock < TRANSITION_BAND_WIDTH * 2]
            if len(near_shock) > 0:
                extra_points = near_shock.sample(min(100, len(near_shock)), random_state=RANDOM_SEED)
                sampled = pd.concat([sampled, extra_points], ignore_index=True)
        
        # Store count of extra near-shock points added (v1 result already captured above)
        base_count = sum(len(df[df['region'] == r]) for r in ['pre', 'post', 'transition']
                         if r != 'post') + min(POST_SHOCK_SAMPLE_GRID**2, len(df[df['region'] == 'post']))
        self.diagnostics['extra_near_shock'] = max(0, len(sampled) - base_count)
        
        return sampled

class Version5_DistanceBased(ShockClassifier):
    """Version 5: Distance-based classification using theoretical shock angle."""
    
    def __init__(self):
        super().__init__("Distance-Based", 5)
        self.shock_angle = None
        self.theoretical_angles = {}
    
    def classify(self, df):
        df = df.copy()
        # Use a Series indexed like df so group assignments land in the right rows
        all_labels = pd.Series(index=df.index, dtype=str)

        for (mach, angle), group in df.groupby(['mach_input', 'angle_input']):
            beta_theory = compute_theoretical_shock_angle(mach, angle)
            self.theoretical_angles[(mach, angle)] = beta_theory

            if not np.isnan(beta_theory):
                slope = np.tan(np.radians(beta_theory))
                y_shock = slope * group['x-coordinate'].values
                y_point = group['y-coordinate'].values

                # Below shock line = near wedge surface = post-shock (compressed)
                # Above shock line = free stream = pre-shock
                group_labels = np.where(y_point < y_shock - TRANSITION_BAND_WIDTH/2, 'post',
                               np.where(y_point > y_shock + TRANSITION_BAND_WIDTH/2, 'pre', 'transition'))
                all_labels[group.index] = group_labels
            else:
                # Fall back to pressure-based (already uses correct convention)
                v1 = Version1_PressureBased()
                all_labels[group.index] = v1.classify(group)

            self.shock_angle = beta_theory

        self.diagnostics = {'shock_angle': self.shock_angle}

        return all_labels.rename('region')
    
    def sample_points(self, df):
        v1 = Version1_PressureBased()
        return v1.sample_points(df)

def process_file(filepath, classifier):
    """Process a single file with the given classifier and enhanced verification."""
    mach, angle = parse_filename(filepath)
    if mach is None:
        return None, None
    
    # Load data
    df = pd.read_excel(filepath, engine='openpyxl')
    df.columns = [c.strip() for c in df.columns]
    
    if 'nodenumber' in df.columns:
        df.drop(columns=['nodenumber'], inplace=True)
    
    # Required columns
    REQUIRED_COLUMNS = ['x-coordinate', 'y-coordinate', 'pressure',
                        'density', 'velocity-magnitude', 'mach-number', 'temperature']
    
    missing_cols = [c for c in REQUIRED_COLUMNS if c not in df.columns]
    if missing_cols:
        return None, None
    
    df = df[REQUIRED_COLUMNS].copy()
    
    # Determine y_max: use actual data max if not hardcoded
    # ANSYS sometimes extends mesh slightly beyond the nominal domain
    y_max_actual = df['y-coordinate'].max()
    y_max_use    = GEOMETRY['y_max'] if GEOMETRY['y_max'] is not None else y_max_actual

    # Filter to geometry bounds
    df = df[(df['x-coordinate'] >= GEOMETRY['x_min']) &
            (df['x-coordinate'] <= GEOMETRY['x_max']) &
            (df['y-coordinate'] >= GEOMETRY['y_min']) &
            (df['y-coordinate'] <= y_max_use)]

    # Log the actual bounds for this file
    print(f"    [{os.path.basename(filepath)}] actual bounds: "
          f"x=[{df['x-coordinate'].min():.4f}, {df['x-coordinate'].max():.4f}]  "
          f"y=[{df['y-coordinate'].min():.4f}, {y_max_actual:.4f}]")
    
    # Clean data
    df.replace([np.inf, -np.inf], np.nan, inplace=True)
    df.dropna(inplace=True)
    
    # Add input parameters
    shock_type = detect_shock_type(filepath)
    df['mach_input']   = mach
    df['angle_input']  = angle
    df['shock_type']   = shock_type   # 'oblique', 'bow', or 'unknown'
    
    # Add geometric features
    # y_over_x is only meaningful away from the apex; clip to a sane range
    x_safe = df['x-coordinate'].clip(lower=1e-4)   # treat anything < 0.1mm as 0.1mm
    df['y_over_x'] = (df['y-coordinate'] / x_safe).clip(-50, 50)
    df['r_from_tip'] = np.sqrt(df['x-coordinate']**2 + df['y-coordinate']**2)
    wedge_slope = np.tan(np.radians(df['angle_input']))
    df['dist_from_wedge'] = df['y-coordinate'] - df['x-coordinate'] * wedge_slope
    df['x_norm'] = df['x-coordinate'] * df['mach_input']
    df['y_norm'] = df['y-coordinate'] * df['mach_input']
    df['polar_angle'] = np.arctan2(df['y-coordinate'], df['x-coordinate'].clip(lower=1e-4))

    # --- Three additional features that directly encode shock geometry ---
    # These help the RF understand where each point sits relative to the shock

    # 1. Theoretical shock angle for this (mach, angle) combination
    beta_theory = compute_theoretical_shock_angle(mach, angle)
    df['shock_angle_theory'] = beta_theory if not np.isnan(beta_theory) else 0.0

    # 2. Signed perpendicular distance from the theoretical shock line
    # Positive = above shock (pre-shock side), Negative = below shock (post-shock side)
    if not np.isnan(beta_theory):
        shock_slope = np.tan(np.radians(beta_theory))
        # Line: y = shock_slope * x, or: shock_slope*x - y = 0
        # Signed distance = (shock_slope*x - y) / sqrt(shock_slope^2 + 1)
        # Positive when above line (pre-shock), negative when below (post-shock)
        # Signed distance: positive = above shock line (pre-shock/freestream side)
        #                  negative = below shock line (post-shock/compressed side)
        # Formula: (y - shock_slope*x) / sqrt(shock_slope^2 + 1)
        numerator = df['y-coordinate'] - shock_slope * df['x-coordinate']
        denominator = np.sqrt(shock_slope**2 + 1.0)
        df['dist_from_shock'] = numerator / denominator
    else:
        df['dist_from_shock'] = 0.0

    # 3. Integer-encoded region label (RF can use this as a categorical split guide)
    # Applied AFTER classify() — so add it after the region assignment below
    
# Classify regions
    df['region'] = classifier.classify(df)

    # Remove per-region statistical outliers BEFORE sampling
    # so that only clean, physically consistent points enter training
    df, outlier_stats = remove_region_outliers(df, z_thresh=3.0)

    # Force x < 0 (upstream of apex) to always be 'pre' — it is pure freestream
    # by definition and no shock can exist there. This overrides any misclassification.
    df.loc[df['x-coordinate'] < 0, 'region'] = 'pre'

# Encode region as integer for RF (0=pre, 1=transition, 2=post)
    region_map = {'pre': 0, 'transition': 1, 'post': 2}
    df['region_encoded'] = df['region'].map(region_map)
    # Sample points
    sampled = classifier.sample_points(df)

    # Extract line profiles for validation
    line_profiles = extract_line_profiles(df, LINE_PROFILES_X, LINE_PROFILES_Y)
    
    # Verify region separation
    verification = verify_region_separation(df, classifier.version_id)
    
# Physics validation via Rankine-Hugoniot jump conditions
    shock_ang = classifier.get_diagnostics().get('shock_angle', None)
    rh_validation = validate_rankine_hugoniot(df, mach, angle, shock_ang)

    # Store diagnostics
    diagnostics = {
        'mach': mach,
        'angle': angle,
        'total_nodes': len(df),
        'classifier_diagnostics': classifier.get_diagnostics(),
        'sampled_nodes': len(sampled),
        'outlier_removal': outlier_stats,
        'rh_validation': rh_validation,
        'verification': verification,
        'line_profiles': line_profiles,
        'filename': os.path.basename(filepath)
    }
    
    return sampled, diagnostics

                sub = fdf[fdf['region'] == reg]
                if len(sub) > 0:
                    ax.scatter(sub['x-coordinate'], sub['y-coordinate'],
                               c=col, s=1, alpha=0.4, label=reg)

            # Overlay shock line
            rh = diag.get('rh_validation', {})
            sa = rh.get('shock_angle_used', None)
            if sa and not np.isnan(sa):
                sl = np.tan(np.radians(sa))
                xs = np.array([0, GEOMETRY['x_max']])
                ax.plot(xs, sl*xs, 'k--', linewidth=1.2, label=f'β={sa:.1f}°')

            ax.set_xlim([GEOMETRY['x_min'], GEOMETRY['x_max']])
            ax.set_ylim([GEOMETRY['y_min'], fdf['y-coordinate'].max() if len(fdf) > 0 else 0.45])
            ax.set_title(f"M={diag['mach']} θ={diag['angle']}°  "
                         f"RH:{'✓' if rh.get('valid') else '✗'}", fontsize=9)
            ax.set_xlabel('x (m)', fontsize=8)
            ax.set_ylabel('y (m)', fontsize=8)
            if idx == 0:
                ax.legend(markerscale=5, fontsize=7)

        for idx in range(n_show, len(axes)):
            axes[idx].set_visible(False)

        plt.tight_layout()
        plt.savefig(os.path.join(version_dir, "per_file_classification.png"), dpi=150)
        plt.close()
    
    print(f"  Diagnostic plots saved to: {version_dir}")

def create_parameter_analysis_plots(version_id, classifier_results_df):
    """Create plots showing relationship between input parameters and shock behavior."""
    analysis_dir = os.path.join(DIAGNOSTICS_DIR, f"version_{version_id}", "parameter_analysis")
    os.makedirs(analysis_dir, exist_ok=True)
    
    # Analyze parameter relationships
    analysis = analyze_parameter_relationship(classifier_results_df)
    
    # Plot 1: Shock angle vs wedge angle for different Mach numbers
    fig, ax = plt.subplots(figsize=(10, 8))
    
    for mach in sorted(analysis['by_mach'].keys()):
        mach_data = classifier_results_df[classifier_results_df['mach'] == mach]
        mach_data = mach_data.dropna(subset=['shock_angle', 'angle'])
        
        if len(mach_data) > 0:
            # Sort by wedge angle
            mach_data_sorted = mach_data.sort_values('angle')
            ax.plot(mach_data_sorted['angle'], mach_data_sorted['shock_angle'], 
                   'o-', label=f'M = {mach}', markersize=4, alpha=0.7)
            
            # Add theoretical curves
            wedge_angles = np.linspace(0, mach_data['angle'].max(), 20)
            theoretical_angles = [compute_theoretical_shock_angle(mach, theta) for theta in wedge_angles]
            ax.plot(wedge_angles, theoretical_angles, '--', alpha=0.5, 
                   label=f'M = {mach} (theoretical)')
    
    ax.set_xlabel('Wedge Angle (degrees)')
    ax.set_ylabel('Shock Angle (degrees)')
    ax.set_title(f'Version {version_id}: Shock Angle vs Wedge Angle')
    ax.grid(alpha=0.3)
    ax.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(analysis_dir, "shock_angle_vs_wedge_angle.png"), dpi=150)
    plt.close()
    
    # Plot 2: Shock angle vs Mach number for different wedge angles
    fig, ax = plt.subplots(figsize=(10, 8))

        angle_data = classifier_results_df[classifier_results_df['angle'] == angle]
        angle_data = angle_data.dropna(subset=['shock_angle', 'mach'])
        
        if len(angle_data) > 0:
            angle_data_sorted = angle_data.sort_values('mach')
            ax.plot(angle_data_sorted['mach'], angle_data_sorted['shock_angle'], 
                   'o-', label=f'θ = {angle}°', markersize=4, alpha=0.7)

    ax.set_ylabel('Shock Angle (degrees)')
    ax.set_title(f'Version {version_id}: Shock Angle vs Mach Number')

    ax.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(analysis_dir, "shock_angle_vs_mach.png"), dpi=150)
    plt.close()
    
    # Plot 3: Correlation heatmap
    if analysis['correlations']:
        fig, ax = plt.subplots(figsize=(8, 6))
        
        # Prepare correlation data
        corr_data = []
        for key, value in analysis['correlations'].items():
            corr_data.append([key.replace('_', ' '), value['correlation'], value['p_value']])
        
        corr_df = pd.DataFrame(corr_data, columns=['Relationship', 'Correlation', 'p-value'])

        bars = ax.barh(corr_df['Relationship'], corr_df['Correlation'],

def save_line_profiles(version_id, all_diagnostics):
    """Save line profile data for ANSYS simulation validation."""
    profiles_dir = os.path.join(LINE_PROFILES_DIR, f"version_{version_id}")
    os.makedirs(profiles_dir, exist_ok=True)
    
    all_profiles = []
    
    for diag in all_diagnostics:
        if 'line_profiles' in diag:
            profiles = diag['line_profiles']
            mach = diag['mach']
            angle = diag['angle']
            
            # Save x-line profiles
            for x_pos, profile in profiles['x_lines'].items():
                profile_df = pd.DataFrame({
                    'y': profile['y'],
                    'pressure': profile['pressure'],
                    'velocity': profile['velocity'],
                    'temperature': profile['temperature'],
                    'density': profile['density']
                })
                profile_df['mach'] = mach
                profile_df['angle'] = angle
                profile_df['x_position'] = x_pos
                profile_df['profile_type'] = 'x_line'
                all_profiles.append(profile_df)
            
            # Save y-line profiles
            for y_pos, profile in profiles['y_lines'].items():
                profile_df = pd.DataFrame({
                    'x': profile['x'],
                    'pressure': profile['pressure'],
                    'velocity': profile['velocity'],
                    'temperature': profile['temperature'],
                    'density': profile['density']
                })
                profile_df['mach'] = mach
                profile_df['angle'] = angle
                profile_df['y_position'] = y_pos
                profile_df['profile_type'] = 'y_line'
                all_profiles.append(profile_df)
    
    if all_profiles:
        # Combine all profiles
        master_profiles = pd.concat(all_profiles, ignore_index=True)
        
        # Save to parquet for efficient storage
        master_profiles.to_parquet(os.path.join(profiles_dir, "all_line_profiles.parquet"), index=False)
        
        # Build summary separately for x-line and y-line profiles
        # (they have different coordinate columns so must not be mixed in one selection)
        summary_parts = []
        x_line_df = master_profiles[master_profiles['profile_type'] == 'x_line']
        y_line_df = master_profiles[master_profiles['profile_type'] == 'y_line']

        if len(x_line_df) > 0:
            summary_parts.append(
                x_line_df[['mach', 'angle', 'profile_type', 'x_position']]
                .drop_duplicates()
                .rename(columns={'x_position': 'position'})
                .assign(axis='x')
            )
        if len(y_line_df) > 0:
            summary_parts.append(
                y_line_df[['mach', 'angle', 'profile_type', 'y_position']]
                .drop_duplicates()
                .rename(columns={'y_position': 'position'})
                .assign(axis='y')
            )

        if summary_parts:
            summary = pd.concat(summary_parts, ignore_index=True)
            summary.to_csv(os.path.join(profiles_dir, "profile_summary.csv"), index=False)
        
        print(f"  Line profiles saved to: {profiles_dir}")
        print(f"    - {len(all_profiles)} profiles extracted")
        if summary_parts:
            print(f"    - {len(summary)} unique profile locations")
        # Plot constant-y line profiles — each shows all 5 variables vs x
        # The shock jump is visible as a step at the shock x-location
        try:
            gamma = 1.4
            for mach_val in sorted(y_line_df['mach'].unique()):
                mach_data = y_line_df[y_line_df['mach'] == mach_val]
                # Pick the median wedge angle case as representative
                angle_val = float(np.median(mach_data['angle'].unique()))
                # Get closest available angle
                available_angles = mach_data['angle'].unique()
                angle_val = available_angles[np.argmin(np.abs(available_angles - angle_val))]
                case_data = mach_data[mach_data['angle'] == angle_val]
                if len(case_data) == 0:
                    continue

                y_positions_found = sorted(case_data['y_position'].unique())
                n_show = min(3, len(y_positions_found))
                if n_show == 0:
                    continue

                fig, axes = plt.subplots(n_show, 5, figsize=(22, 4 * n_show))
                if n_show == 1:
                    axes = axes.reshape(1, -1)

                plot_vars = [
                    ('pressure',    'Pressure (Pa)',     'red'),
                    ('density',     'Density (kg/m³)',   'purple'),
                    ('velocity',    'Velocity (m/s)',    'blue'),
                    ('temperature', 'Temperature (K)',   'orange'),
                ]

                for row_idx, yp in enumerate(y_positions_found[:n_show]):
                    slice_data = case_data[np.isclose(case_data['y_position'], yp, atol=0.003)]
                    if len(slice_data) == 0:
                        continue
                    slice_sorted = slice_data.sort_values('x')

                    # Compute theoretical shock x-location at this y
                    # x_shock = y / tan(beta_theory)
                    beta_theory = compute_theoretical_shock_angle(mach_val, angle_val)
                    x_shock_theory = yp / np.tan(np.radians(beta_theory)) if not np.isnan(beta_theory) else None

                    # Freestream reference values (minimum pressure = pre-shock)
                    P_fs   = slice_sorted['pressure'].min()
                    T_fs   = slice_sorted['temperature'].min()
                    rho_fs = slice_sorted['density'].min()

                    for col_idx, (col, ylabel, color) in enumerate(plot_vars):
                        ax = axes[row_idx, col_idx]
                        if col not in slice_sorted.columns:
                            continue
                        ax.plot(slice_sorted['x'].values, slice_sorted[col].values,
                                color=color, linewidth=1.5, label='CFD data')

                        # Freestream reference line
                        fs_ref = {'pressure': P_fs, 'density': rho_fs,
                                  'temperature': T_fs, 'velocity': slice_sorted['velocity'].max()}
                        if col in fs_ref:
                            ax.axhline(fs_ref[col], color='navy', linestyle='--',
                                       alpha=0.6, linewidth=1, label='Freestream')

                        # Theoretical shock x-location
                        if x_shock_theory is not None:
                            ax.axvline(x_shock_theory, color='green', linestyle='-.',
                                       alpha=0.8, linewidth=1.2, label='Shock (theory)')

                        ax.set_xlabel('x (m)')
                        ax.set_ylabel(ylabel)
                        ax.set_title(f'y = {yp:.3f} m', fontsize=9)
                        ax.grid(alpha=0.3)
                        if row_idx == 0 and col_idx == 0:
                            ax.legend(fontsize=7)

                    # 5th column: derived local Mach number
                    ax5 = axes[row_idx, 4]
                    if 'velocity' in slice_sorted.columns and 'temperature' in slice_sorted.columns:
                        a_local = np.sqrt(gamma * 287.0 * slice_sorted['temperature'].values)
                        M_local = slice_sorted['velocity'].values / a_local
                        ax5.plot(slice_sorted['x'].values, M_local, color='green', linewidth=1.5, label='CFD data')
                        ax5.axhline(mach_val, color='navy', linestyle='--', alpha=0.6, linewidth=1, label='Freestream M')
                        if x_shock_theory is not None:
                            ax5.axvline(x_shock_theory, color='green', linestyle='-.',
                                        alpha=0.8, linewidth=1.2)
                        ax5.set_xlabel('x (m)')
                        ax5.set_ylabel('Local Mach')
                        ax5.set_title(f'y = {yp:.3f} m', fontsize=9)
                        ax5.grid(alpha=0.3)

                fig.suptitle(f'Line profiles at constant y — M={mach_val}, θ={angle_val}°\n'
                             f'(step jump marks shock location; green line = theoretical shock x)',
                             fontsize=11)
                plt.tight_layout()
                fname = f"line_profiles_y_M{mach_val}_angle{angle_val}.png"
                plt.savefig(os.path.join(profiles_dir, fname), dpi=150, bbox_inches='tight')
                plt.close()
                print(f"    Saved line profile plot: {fname}")

        except Exception as e:
            print(f"    Warning: line profile plot failed: {e}")

def generate_code1_blackbox(output_dir, versions_dir, verification_dir):
    """
    Code I Black Box — comprehensive text diagnostic report.
    Reads all CSV/parquet outputs from Code I and summarises everything
    needed to verify preprocessing quality before running Code II.

    Saves: {output_dir}/blackbox_code1_report.txt
    Download this single file after Code I finishes.
    """
    import os
    import numpy as np
    import pandas as pd
    from scipy.optimize import brentq
    from datetime import datetime

    lines = []
    def w(s=""): lines.append(str(s))

    def compute_beta(mach, wedge_deg):
        gamma = 1.4
        mu = np.degrees(np.arcsin(1.0 / mach))
        def sr(b):
            br = np.radians(b)
            return (np.tan(np.radians(wedge_deg)) -
                    2/np.tan(br) * (mach**2*np.sin(br)**2 - 1) /
                    (mach**2*(gamma + np.cos(2*br)) + 2))
        bv = np.linspace(mu + 0.5, 89, 500)
        sv = [sr(b) for b in bv]
        sc = [(bv[i], bv[i+1]) for i in range(len(sv)-1) if sv[i]*sv[i+1] < 0]
        if not sc:
            return np.nan
        try:
            return brentq(sr, sc[0][0], sc[0][1])
        except Exception:
            return np.nan

    w("=" * 78)
    w("  OBLIQUE SHOCK ML — CODE I PREPROCESSING BLACK BOX REPORT")
    w("  Generated automatically at end of Code I preprocessing")
    w("=" * 78)
    w()

    # ── Section 0: Reproducibility stamp ────────────────────────────────
    w("── SECTION 0: REPRODUCIBILITY STAMP ──────────────────────────────────────")
    w(f"  Date generated:       {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    w(f"  Random seed:          {RANDOM_SEED}")
    w(f"  Geometry x:           [{GEOMETRY['x_min']}, {GEOMETRY['x_max']}] m")
    w(f"  Geometry y:           [{GEOMETRY['y_min']}, auto-detected] m")
    w(f"  Wedge apex:           ({GEOMETRY['wedge_apex_x']}, {GEOMETRY['wedge_apex_y']})")
    w(f"  Transition band:      {TRANSITION_BAND_WIDTH} m  (±{TRANSITION_BAND_WIDTH/2} m each side)")
    w(f"  Pre-shock sample:     {PRE_SHOCK_SAMPLE_FRAC*100:.1f}% of pre-shock points")
    w(f"  Post-shock grid:      {POST_SHOCK_SAMPLE_GRID}×{POST_SHOCK_SAMPLE_GRID}")
    w(f"  Outlier removal:      z-score threshold = 3.0 (pre: 4.5)")
    w(f"  Input directory:      {INPUT_DIR}")
    w()

    # ── SECTION 1: Version comparison (R-H pass rates) ──────────────────────
    w("── SECTION 1: VERSION COMPARISON ─────────────────────────────────────────")
    w()
    comp_csv = os.path.join(verification_dir, "versions_comparison.csv")
    if os.path.exists(comp_csv):
        comp = pd.read_csv(comp_csv)
        w(comp.to_string(index=False))
        w()
        # Identify best version
        if 'rh_passed' in comp.columns:
            best = comp.loc[comp['rh_passed'].astype(str).str.extract(r'(\d+)')[0].astype(int).idxmax()]
            w(f"  Best version by R-H: Version {best['version']}")
        w()
    else:
        w("  versions_comparison.csv not found")
    w()

    # ── SECTION 2: V3 parquet sanity check (Code II input) ──────────────────
    w("── SECTION 2: VERSION 3 PARQUET SANITY CHECK (Code II input) ─────────────")
    w()
    v3_path = os.path.join(versions_dir, "version_3_full.parquet")
    if os.path.exists(v3_path):
        df = pd.read_parquet(v3_path)
        w(f"  Shape:   {df.shape[0]:,} rows × {df.shape[1]} columns")
        w(f"  Columns: {list(df.columns)}")
        w()

        # NaN / Inf
        num = df.select_dtypes(include=[np.number])
        nan_total = int(num.isna().sum().sum())
        inf_total = int(np.isinf(num.values).sum())
        w(f"  NaN total: {nan_total}  ({'✓ clean' if nan_total==0 else '✗ PROBLEM'})")
        w(f"  Inf total: {inf_total}  ({'✓ clean' if inf_total==0 else '✗ PROBLEM'})")
        w()

        # Geometry
        w(f"  Geometry:")
        w(f"    x: [{df['x-coordinate'].min():.4f}, {df['x-coordinate'].max():.4f}]  "
          f"{'✓' if df['x-coordinate'].min()>=-0.101 and df['x-coordinate'].max()<=0.301 else '✗'}")
        w(f"    y: [{df['y-coordinate'].min():.4f}, {df['y-coordinate'].max():.4f}]  "
          f"{'✓' if df['y-coordinate'].min()>=-0.001 and df['y-coordinate'].max()<=0.461 else '✗'}")
        w()

        # Cases
        cases = df.groupby(['mach_input', 'angle_input']).size()
        w(f"  Cases: {len(cases)} found")
        if 'shock_type' in df.columns:

        w()

        # Shock type breakdown
        if 'shock_type' in df.columns:
            w("  Shock type breakdown:")
            for st, stg in df.groupby('shock_type'):

                w(f"    {st:10s}: {n_cases_st} cases  {n_pts_st:,} points "

            w()

        # Outlier removal summary (read from diagnostics CSV)
        diag_csv_s2 = os.path.join(versions_dir, "version_3_diagnostics.csv")
        if os.path.exists(diag_csv_s2):
            import ast
            diag_s2 = pd.read_csv(diag_csv_s2)
            if 'outlier_removal' in diag_s2.columns:
                w("  Outlier removal summary (z-score=3.0 per region):")
                w(f"  {'M':>5}  {'θ':>6}  "
                  f"{'pre_removed':>12}  {'trans_removed':>14}  "
                  f"{'post_removed':>13}")
                w("  " + "─" * 55)
                total_removed = {'pre': 0, 'transition': 0, 'post': 0}
                for _, row in diag_s2.iterrows():
                    try:
