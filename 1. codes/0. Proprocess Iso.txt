def parse_filename_raw(fname):
    fname = os.path.basename(fname)

    # wedge_M2_0_angle12_5.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_(\d+)_angle(\d+)_(\d+)', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), float(f"{m.group(3)}.{m.group(4)}")

    # wedge_M2_0_angle7.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_(\d+)_angle(\d+)(?:[^\d_]|$)', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), float(m.group(3))

    # wedge_M2.5_angle10.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+\.\d+)_angle(\d+(?:\.\d+)?)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), float(m.group(2))

    # wedge_M3_angle10.xlsx
    m = re.search(r'(?:wedge|oblique)_M(\d+)_angle(\d+(?:\.\d+)?)', fname, re.IGNORECASE)
    if m:
        return float(m.group(1)), float(m.group(2))

    # Generic fallback
    m = re.search(r'_M(\d+)_(\d+)_', fname, re.IGNORECASE)
    if m:
        return float(f"{m.group(1)}.{m.group(2)}"), 0.0

    return None, None


def parse_filename(fname):
    mach, angle = parse_filename_raw(fname)
    if mach is None:
        return None, None
    return round(mach, 6), round(angle, 6)


def compute_theoretical_shock_angle(mach, wedge_angle_deg):
    """
    Solve the θ-β-M relation for an oblique shock (weak solution).
    Returns shock angle β in degrees, or NaN if detached / no solution.
    """
    if wedge_angle_deg <= 0:
        return np.nan
    gamma = 1.4
    try:
        mach_angle = np.degrees(np.arcsin(1.0 / mach))
    except (ValueError, ZeroDivisionError):
        return np.nan

    def shock_relation(beta_deg):
        beta_rad  = np.radians(beta_deg)
        wedge_rad = np.radians(wedge_angle_deg)
        tan_theta = (2 / np.tan(beta_rad) *
                     (mach**2 * np.sin(beta_rad)**2 - 1) /
                     (mach**2 * (gamma + np.cos(2 * beta_rad)) + 2))
        return np.tan(wedge_rad) - tan_theta

    bv = np.linspace(mach_angle + 0.5, 89, 500)
    sv = [shock_relation(b) for b in bv]
    sc = [(bv[i], bv[i+1]) for i in range(len(sv)-1) if sv[i]*sv[i+1] < 0]
    if not sc:
        return np.nan
    try:
        return brentq(shock_relation, sc[0][0], sc[0][1])
    except Exception:
        return np.nan


def fit_shock_line(points, slope_only=True):
    """
    Fit a line through shock candidate points using RANSAC.
    When slope_only=True, the intercept is forced to zero
    (shock must pass through the wedge apex at the origin).

    Returns (slope, intercept, angle_deg) or (None, None, None).
    """
    if len(points) < 2:
        return None, None, None

    X = points[:, 0].reshape(-1, 1)
    y = points[:, 1]

    ransac = RANSACRegressor(random_state=RANDOM_SEED, residual_threshold=0.01)
    ransac.fit(X, y)

    if slope_only:
        # Origin-forced least-squares slope
        slope     = np.sum(X.flatten() * y) / (np.sum(X.flatten()**2) + 1e-10)
        intercept = 0.0
    else:
        slope     = ransac.estimator_.coef_[0]
        intercept = ransac.estimator_.intercept_

    angle_deg = np.degrees(np.arctan(slope))
    return slope, intercept, angle_deg


def remove_region_outliers(df, z_thresh=3.0):
    """
    Remove physically implausible points within each region using z-score.
    Pre-shock outlier threshold is looser (z_thresh × 1.5) because
    freestream is genuinely uniform — any deviation is almost certainly
    a mesh artefact.

    Returns (clean_df, removal_stats).
    """
    FLOW_VARS = ['pressure', 'density', 'temperature', 'velocity-magnitude', 'mach-number']
    clean_parts  = []
    removal_stats = {}

    for region in ['pre', 'post', 'transition']:
        region_df = df[df['region'] == region].copy()
        if len(region_df) == 0:
            continue

        thresh       = z_thresh * 1.5 if region == 'pre' else z_thresh
        outlier_mask = pd.Series(False, index=region_df.index)

        for var in FLOW_VARS:
            if var not in region_df.columns:
                continue
            vals = region_df[var]
            mean, std = vals.mean(), vals.std()
            if std < 1e-10:
                continue
            outlier_mask |= (np.abs((vals - mean) / std) > thresh)

        n_before = len(region_df)
        clean_region = region_df[~outlier_mask]
        n_removed    = n_before - len(clean_region)

        removal_stats[region] = {
            'before': n_before,
            'removed': n_removed,
            'fraction_removed': n_removed / n_before if n_before > 0 else 0,
        }
        clean_parts.append(clean_region)

    clean_df = pd.concat(clean_parts, ignore_index=False)
    return clean_df, removal_stats


def validate_rankine_hugoniot(df, mach, wedge_angle_deg, shock_angle_deg=None):
    """
    Validate region classification against Rankine-Hugoniot jump conditions.
    Returns a dict with ratios, theoretical values, and pass/fail flags.
    Tolerance: 5% on each of P, T, rho ratios.
    """
    gamma = 1.4
    pre  = df[df['region'] == 'pre']
    post = df[df['region'] == 'post']

    if len(pre) == 0 or len(post) == 0:
        return {'valid': False, 'reason': 'missing region'}
    if wedge_angle_deg <= 0:
        return {'valid': False, 'reason': 'bow shock — no single beta'}

    p1,  T1,  rho1 = pre['pressure'].mean(),  pre['temperature'].mean(),  pre['density'].mean()
    p2,  T2,  rho2 = post['pressure'].mean(), post['temperature'].mean(), post['density'].mean()

    # Determine shock angle to use
    if shock_angle_deg is None or (isinstance(shock_angle_deg, float) and np.isnan(shock_angle_deg)):
        shock_angle_deg = compute_theoretical_shock_angle(mach, wedge_angle_deg)

    # If analytical solution also failed, invert pressure ratio
    if shock_angle_deg is None or (isinstance(shock_angle_deg, float) and np.isnan(shock_angle_deg)):
        p_ratio_measured = p2 / p1
        Mn1_sq = (p_ratio_measured * (gamma + 1) + (gamma - 1)) / (2 * gamma)
        if Mn1_sq > 0:
            shock_angle_deg = np.degrees(np.arcsin(
                np.clip(np.sqrt(Mn1_sq) / mach, 0, 1)
            ))
        else:
            return {'valid': False, 'reason': 'cannot determine shock angle'}

    beta_rad = np.radians(shock_angle_deg)
    Mn1      = mach * np.sin(beta_rad)

    p_ratio_theory   = (2 * gamma * Mn1**2 - (gamma - 1)) / (gamma + 1)
    T_ratio_theory   = ((2 * gamma * Mn1**2 - (gamma-1)) *
                        ((gamma-1) * Mn1**2 + 2)) / ((gamma+1)**2 * Mn1**2)
    rho_ratio_theory = (gamma+1) * Mn1**2 / ((gamma-1) * Mn1**2 + 2)

    p_ratio_data   = p2 / p1
    T_ratio_data   = T2 / T1
    rho_ratio_data = rho2 / rho1

    tol   = 0.05
    p_ok  = abs(p_ratio_data   - p_ratio_theory)   / p_ratio_theory   < tol
    T_ok  = abs(T_ratio_data   - T_ratio_theory)   / T_ratio_theory   < tol
    rho_ok= abs(rho_ratio_data - rho_ratio_theory) / rho_ratio_theory < tol

    return {
        'valid':            p_ok and T_ok and rho_ok,
        'shock_angle_used': shock_angle_deg,
        'Mn1':              Mn1,
        'p_ratio_data':     p_ratio_data,   'p_ratio_theory':   p_ratio_theory,   'p_ok':   p_ok,
        'T_ratio_data':     T_ratio_data,   'T_ratio_theory':   T_ratio_theory,   'T_ok':   T_ok,
        'rho_ratio_data':   rho_ratio_data, 'rho_ratio_theory': rho_ratio_theory, 'rho_ok': rho_ok,
        'p_error_pct':    abs(p_ratio_data   - p_ratio_theory)   / p_ratio_theory   * 100,
        'T_error_pct':    abs(T_ratio_data   - T_ratio_theory)   / T_ratio_theory   * 100,
        'rho_error_pct':  abs(rho_ratio_data - rho_ratio_theory) / rho_ratio_theory * 100,
    }


def verify_region_separation(df):
    """
    Run two spatial and statistical tests to confirm regions are well-separated.
    Returns a verification dict consumed by diagnostics and the black-box report.
    """
    verification = {'tests': {}}

    # Test 1 — KS test: pre and post pressure distributions should be distinct
    pre_p  = df[df['region'] == 'pre']['pressure'].values
    post_p = df[df['region'] == 'post']['pressure'].values

    if len(pre_p) > 0 and len(post_p) > 0:
        ks_stat, ks_pval = ks_2samp(pre_p, post_p)
        verification['tests']['pressure_distribution'] = {
            'ks_statistic': ks_stat,
            'ks_pvalue':    ks_pval,
            'pre_mean':     pre_p.mean(),
            'post_mean':    post_p.mean(),
            'distinct':     ks_stat > 0.3 and ks_pval < 0.05,
        }

    # Test 2 — Spatial separation along shock-normal axis
    pre_df  = df[df['region'] == 'pre']
    post_df = df[df['region'] == 'post']

    if len(pre_df) > 0 and len(post_df) > 0:
        pre_cx,  pre_cy  = pre_df['x-coordinate'].mean(),  pre_df['y-coordinate'].mean()
        post_cx, post_cy = post_df['x-coordinate'].mean(), post_df['y-coordinate'].mean()
        dx, dy = post_cx - pre_cx, post_cy - pre_cy
        norm   = np.sqrt(dx**2 + dy**2) + 1e-10
        nx, ny = dx / norm, dy / norm

        pre_proj  = pre_df['x-coordinate'].values  * nx + pre_df['y-coordinate'].values  * ny
        post_proj = post_df['x-coordinate'].values * nx + post_df['y-coordinate'].values * ny

        gap = post_proj.min() - pre_proj.max()
        verification['tests']['spatial_separation'] = {
            'pre_proj_range':  (float(pre_proj.min()),  float(pre_proj.max())),
            'post_proj_range': (float(post_proj.min()), float(post_proj.max())),
            'separated':       gap > 0,
            'gap':             float(gap),
        }
    else:
        verification['tests']['spatial_separation'] = {
            'separated': False, 'gap': 0.0,
        }

    # Test 3 — Transition region point count
    trans = df[df['region'] == 'transition']
    if len(trans) > 0:
        verification['tests']['transition_region'] = {
            'num_points': len(trans),
            'y_range':    (trans['y-coordinate'].min(), trans['y-coordinate'].max()),
            'width':      trans['y-coordinate'].max() - trans['y-coordinate'].min(),
        }

    return verification


def extract_line_profiles(df, x_positions, y_positions):
    """Extract pressure/velocity/temperature/density along fixed x and y slices."""
    profiles = {'x_lines': {}, 'y_lines': {}}
    tol = 0.002

    for x in x_positions:
        mask = np.abs(df['x-coordinate'] - x) < tol
        if mask.any():
            sub = df[mask].sort_values('y-coordinate')
            profiles['x_lines'][x] = {
                'y':           sub['y-coordinate'].values,
                'pressure':    sub['pressure'].values,
                'velocity':    sub['velocity-magnitude'].values,
                'temperature': sub['temperature'].values,
                'density':     sub['density'].values,
            }

    for y in y_positions:
        mask = np.abs(df['y-coordinate'] - y) < tol
        if mask.any():
            sub = df[mask].sort_values('x-coordinate')
            profiles['y_lines'][y] = {
                'x':           sub['x-coordinate'].values,
                'pressure':    sub['pressure'].values,
                'velocity':    sub['velocity-magnitude'].values,
                'temperature': sub['temperature'].values,
                'density':     sub['density'].values,
            }

    return profiles

class Version1_PressureBased:
    """
    Pressure-based bootstrap classifier used internally by Version 3.
    Classifies by pressure relative to the midpoint of the observed
    pressure range ±10% of the total jump.
    """

    def __init__(self):
        self.diagnostics = {}

    def classify(self, df):
        p     = df['pressure'].values
        p_min, p_max = p.min(), p.max()
        jump  = p_max - p_min
        p_mid = p_min + 0.5 * jump
        band  = 0.10 * jump

        pre_threshold  = p_mid - band
        post_threshold = p_mid + band

        labels = np.where(p <= pre_threshold, 'pre',
                 np.where(p >= post_threshold, 'post', 'transition'))

        self.diagnostics = {
            'p_min': p_min, 'p_max': p_max,
            'p_mid': p_mid, 'band_width': band,
            'pre_threshold':  pre_threshold,
            'post_threshold': post_threshold,
        }
        return pd.Series(labels, index=df.index, name='region')

    def sample_points(self, df):
        """
        Stratified sampler used after classification.
        Pre-shock: 2% random sample (large uniform region — dense sampling wasteful).
        Transition: keep all points (critical for shock learning).
        Post-shock: grid-based spatial sample to ensure coverage.
        """
        pre_df   = df[df['region'] == 'pre']
        post_df  = df[df['region'] == 'post']
        trans_df = df[df['region'] == 'transition']

        pre_sample   = pre_df.sample(frac=PRE_SHOCK_SAMPLE_FRAC, random_state=RANDOM_SEED)
        trans_sample = trans_df  # keep all

        if len(post_df) > 0:
            x_edges = np.linspace(post_df['x-coordinate'].min(),
                                  post_df['x-coordinate'].max(),
                                  POST_SHOCK_SAMPLE_GRID)
            y_edges = np.linspace(post_df['y-coordinate'].min(),
                                  post_df['y-coordinate'].max(),
                                  POST_SHOCK_SAMPLE_GRID)

            post_sample_idx = []
            for i in range(len(x_edges) - 1):
                for j in range(len(y_edges) - 1):
                    cell = post_df[
                        (post_df['x-coordinate'] >= x_edges[i]) &
                        (post_df['x-coordinate'] <  x_edges[i+1]) &
                        (post_df['y-coordinate'] >= y_edges[j]) &
                        (post_df['y-coordinate'] <  y_edges[j+1])
                    ]
                    if len(cell) > 0:
                        post_sample_idx.append(cell.sample(1).index[0])

            post_sample = (post_df.loc[post_sample_idx] if post_sample_idx
                           else post_df.sample(n=min(500, len(post_df)),
                                               random_state=RANDOM_SEED))
            # Guarantee minimum density
            if len(post_sample) < 200 and len(post_df) >= 200:
                post_sample = post_df.sample(n=200, random_state=RANDOM_SEED)
        else:
            post_sample = post_df

        return pd.concat([pre_sample, post_sample, trans_sample], ignore_index=True)

class Version3_ShockLineFitting:
    """
    Two-stage classifier:
      Stage 1 — Bootstrap with pressure-based classification (V1).
      Stage 2 — Fit an origin-forced shock line through the transition
                 candidates; reclassify all points by signed distance
                 from that line.

    Falls back to Version 1 if fewer than 10 transition candidates
    are found or if the line fit fails.
    """

    def __init__(self):
        self.shock_slope     = None
        self.shock_intercept = None
        self.shock_angle     = None
        self.diagnostics     = {}

    def get_diagnostics(self):
        return self.diagnostics

    def classify(self, df):
        # Stage 1: pressure bootstrap
        v1      = Version1_PressureBased()
        regions = v1.classify(df)
        df      = df.copy()
        df['region_temp'] = regions

        candidate_df = df[df['region_temp'] == 'transition']

        if len(candidate_df) >= 10:
            candidate_coords = candidate_df[['x-coordinate', 'y-coordinate']].values
            slope, intercept, angle = fit_shock_line(candidate_coords, slope_only=True)

            if slope is not None:
                self.shock_slope     = slope
                self.shock_intercept = intercept
                self.shock_angle     = angle

                # Stage 2: reclassify by distance from fitted shock line
                y_shock = slope * df['x-coordinate'].values + intercept
                y_point = df['y-coordinate'].values
                half    = TRANSITION_BAND_WIDTH / 2

                labels = np.where(y_point < y_shock - half, 'post',
                         np.where(y_point > y_shock + half, 'pre', 'transition'))

                self.diagnostics = {
                    'shock_slope':        slope,
                    'shock_intercept':    intercept,
                    'shock_angle':        angle,
                    'candidate_points':   len(candidate_df),
                    'fallback_used':      False,
                }
                return pd.Series(labels, index=df.index, name='region')

        # Fallback
        print("      ⚠ V3 fallback to V1 (insufficient candidates or fit failed)")
        self.diagnostics = {
            'shock_slope':       None,
            'shock_intercept':   None,
            'shock_angle':       None,
            'candidate_points':  len(candidate_df),
            'fallback_used':     True,
        }
        return v1.classify(df)

    def sample_points(self, df):
        return Version1_PressureBased().sample_points(df)

def process_file(filepath):
    """
    Load, clean, classify with V3, and compute all diagnostics
    for a single Excel file.

    Returns (sampled_df, diagnostics_dict) or (None, None) on failure.
    """
    mach, angle = parse_filename(filepath)
    if mach is None:
        print(f"  ⚠ Could not parse filename: {os.path.basename(filepath)}")
        return None, None

    # Load
    try:
        df = pd.read_excel(filepath, engine='openpyxl')
    except Exception as e:
        print(f"  ✗ Failed to read {os.path.basename(filepath)}: {e}")
        return None, None

    df.columns = [c.strip() for c in df.columns]
    if 'nodenumber' in df.columns:
        df.drop(columns=['nodenumber'], inplace=True)

    REQUIRED = ['x-coordinate', 'y-coordinate', 'pressure',
                'density', 'velocity-magnitude', 'mach-number', 'temperature']
    missing = [c for c in REQUIRED if c not in df.columns]
    if missing:
        print(f"  ✗ Missing columns in {os.path.basename(filepath)}: {missing}")
        return None, None

    df = df[REQUIRED].copy()

    # Filter to geometry bounds
    y_max_actual = df['y-coordinate'].max()
    y_max_use    = GEOMETRY['y_max'] if GEOMETRY['y_max'] is not None else y_max_actual
    df = df[
        (df['x-coordinate'] >= GEOMETRY['x_min']) &
        (df['x-coordinate'] <= GEOMETRY['x_max']) &
        (df['y-coordinate'] >= GEOMETRY['y_min']) &
        (df['y-coordinate'] <= y_max_use)
    ]
    print(f"    [{os.path.basename(filepath)}] "
          f"x=[{df['x-coordinate'].min():.4f}, {df['x-coordinate'].max():.4f}]  "
          f"y=[{df['y-coordinate'].min():.4f}, {y_max_actual:.4f}]")

    # Clean
    df.replace([np.inf, -np.inf], np.nan, inplace=True)
    df.dropna(inplace=True)

    # Input parameters
    df['mach_input']  = mach
    df['angle_input'] = angle

    # Geometric features
    x_safe = df['x-coordinate'].clip(lower=1e-4)
    df['y_over_x']         = (df['y-coordinate'] / x_safe).clip(-50, 50)
    df['r_from_tip']       = np.sqrt(df['x-coordinate']**2 + df['y-coordinate']**2)
    wedge_slope_feat       = np.tan(np.radians(df['angle_input']))
    df['dist_from_wedge']  = df['y-coordinate'] - df['x-coordinate'] * wedge_slope_feat
    df['x_norm']           = df['x-coordinate'] * df['mach_input']
    df['y_norm']           = df['y-coordinate'] * df['mach_input']
    df['polar_angle']      = np.arctan2(df['y-coordinate'],
                                        df['x-coordinate'].clip(lower=1e-4))

    # Physics features from theoretical shock angle
    beta_theory = compute_theoretical_shock_angle(mach, angle)
    df['shock_angle_theory'] = beta_theory if not np.isnan(beta_theory) else 0.0

    if not np.isnan(beta_theory):
        shock_slope_feat = np.tan(np.radians(beta_theory))
        numerator        = df['y-coordinate'] - shock_slope_feat * df['x-coordinate']
        denominator      = np.sqrt(shock_slope_feat**2 + 1.0)
        df['dist_from_shock'] = numerator / denominator
    else:
        df['dist_from_shock'] = 0.0

    # Classify with V3
    classifier = Version3_ShockLineFitting()
    df['region'] = classifier.classify(df)

    # Outlier removal
    df, outlier_stats = remove_region_outliers(df, z_thresh=3.0)

    # Force x < 0 to pre-shock (pure freestream by definition)
    df.loc[df['x-coordinate'] < 0, 'region'] = 'pre'

    # Integer-encode region
    region_map = {'pre': 0, 'transition': 1, 'post': 2}
    df['region_encoded'] = df['region'].map(region_map)

    # Sample
    sampled = classifier.sample_points(df)

    # Diagnostics
    line_profiles = extract_line_profiles(df, LINE_PROFILES_X, LINE_PROFILES_Y)
    verification  = verify_region_separation(df)
    rh_validation = validate_rankine_hugoniot(
        df, mach, angle,
        classifier.get_diagnostics().get('shock_angle', None)
    )

    diagnostics = {
        'mach':                   mach,
        'angle':                  angle,
        'total_nodes':            len(df),
        'sampled_nodes':          len(sampled),
        'classifier_diagnostics': classifier.get_diagnostics(),
        'outlier_removal':        outlier_stats,
        'rh_validation':          rh_validation,
        'verification':           verification,
        'line_profiles':          line_profiles,
        'filename':               os.path.basename(filepath),
    }

    return sampled, diagnostics

# MAIN

def main():
    print("=" * 70)
    print("  DATA PREPROCESSING — VERSION 3 (SHOCK LINE FITTING) ONLY")
    print("=" * 70)
    print(f"\nGeometry: x=[{GEOMETRY['x_min']}, {GEOMETRY['x_max']}]  "
          f"y=[{GEOMETRY['y_min']}, {GEOMETRY['y_max']}]")
    print(f"Transition band: ±{TRANSITION_BAND_WIDTH/2} m\n")

    # Discover and validate Excel files
    all_xlsx = sorted([
        os.path.join(INPUT_DIR, f)
        for f in os.listdir(INPUT_DIR)
        if f.lower().endswith('.xlsx')
    ])

    excel_files, bad_files = [], []
    for fpath in all_xlsx:
        mach, angle = parse_filename(fpath)
        if mach is None:
            bad_files.append(os.path.basename(fpath))
        else:
            excel_files.append(fpath)

    if bad_files:
        print("\n" + "=" * 70)
        print("  ❌  UNRECOGNISED FILENAME(S) — CANNOT PROCEED")
        print("=" * 70)
        for bf in bad_files:
            print(f"    ✗  {bf}")
        print("""
  Expected format:
    wedge_M<mach>_angle<angle>.xlsx
    wedge_M3_5_angle22_5.xlsx  (underscores as decimal separator)
    wedge_M3.5_angle22.5.xlsx  (dots also accepted)
  Please rename and re-run.
""")
        return

    if not excel_files:
        print("❌ No valid Excel files found. Check INPUT_DIR.")
        return

    print(f"Found {len(excel_files)} valid Excel files\n")

    # Process all files
    all_sampled          = []
    version_diagnostics  = []
    classifier_results   = []

    for filepath in tqdm(excel_files, desc="V3 processing"):
        sampled, diag = process_file(filepath)
        if sampled is not None:
            all_sampled.append(sampled)
            version_diagnostics.append(diag)

            cd = diag.get('classifier_diagnostics', {})
            # Preferred detected angle; fall back to theoretical
            detected_angle = cd.get('shock_angle', None)
            if detected_angle is None or (isinstance(detected_angle, float) and
                                          np.isnan(detected_angle)):
                rh = diag.get('rh_validation', {})
                detected_angle = rh.get('shock_angle_used', np.nan)
            if detected_angle is None or (isinstance(detected_angle, float) and
                                          np.isnan(detected_angle)):
                detected_angle = compute_theoretical_shock_angle(diag['mach'], diag['angle'])

            classifier_results.append({
                'mach':        diag['mach'],
                'angle':       diag['angle'],
                'shock_angle': detected_angle,
                'fallback':    cd.get('fallback_used', False),
                'filename':    diag.get('filename', ''),
            })

    if not all_sampled:
        print("❌ No data processed. Check INPUT_DIR and filenames.")
        return

    # Combine
    master_df = pd.concat(all_sampled, ignore_index=True)
    master_df.replace([np.inf, -np.inf], np.nan, inplace=True)
    master_df.dropna(inplace=True)

    # Save outputs
    full_path  = os.path.join(VERSIONS_DIR, "version_3_full.parquet")
    train_path = os.path.join(VERSIONS_DIR, "version_3.parquet")
    master_df.to_parquet(full_path, index=False)
    master_df.drop(columns=['region']).to_parquet(train_path, index=False)

    diag_df = pd.DataFrame(version_diagnostics)
    diag_df.to_csv(os.path.join(VERSIONS_DIR, "version_3_diagnostics.csv"), index=False)

    params_df = pd.DataFrame(classifier_results)
    params_df.to_csv(os.path.join(VERSIONS_DIR, "version_3_parameters.csv"), index=False)

    # Output means per region
