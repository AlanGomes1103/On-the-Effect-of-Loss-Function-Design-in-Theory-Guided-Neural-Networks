def compute_shock_angle(mach, wedge_deg):
    """Solve θ-β-M relation for weak shock. Returns β in degrees, or NaN if detached."""
    try:
        mu = np.degrees(np.arcsin(1.0 / mach))
    except (ValueError, ZeroDivisionError):
        return np.nan

    def residual(b):
        br = np.radians(b)
        return (np.tan(np.radians(wedge_deg)) -
                2.0 / np.tan(br) *
                (mach**2 * np.sin(br)**2 - 1.0) /
                (mach**2 * (GAMMA + np.cos(2 * br)) + 2.0))

    bv = np.linspace(mu + 0.5, 89.0, 800)
    sv = [residual(b) for b in bv]
    crossings = [(bv[i], bv[i+1]) for i in range(len(sv) - 1)
                 if sv[i] * sv[i+1] < 0]
    if not crossings:
        return np.nan
    try:
        return brentq(residual, crossings[0][0], crossings[0][1])
    except Exception:
        return np.nan

def beta_angle(XX, YY, pred_dict, fluid_mask, beta_deg):
    """Return actual shock angle from grid."""
    beta_angles = []
    for target in ['pressure', 'mach-number']:
        Z = pred_dict[target]                          # 2D array (grid_res x grid_res)
        # Compute gradient magnitude (only on fluid region)
        grad_x = np.gradient(Z, axis=1)
        grad_y = np.gradient(Z, axis=0)
        grad_mag = np.sqrt(grad_x**2 + grad_y**2)
        grad_mag[~fluid_mask] = np.nan                # mask out wedge/non‑fluid

        # Threshold using high percentile of valid gradients
        finite = grad_mag[np.isfinite(grad_mag)]
        if len(finite) < 10:
            continue
        threshold = np.nanpercentile(finite, 97.0)
        shock_mask = grad_mag > threshold
        shock_indices = np.where(shock_mask)
        if len(shock_indices[0]) < 20:
            continue

        # Get physical coordinates of shock points
        shock_x = XX[shock_indices]
        shock_y = YY[shock_indices]
        # Fit line y = m*x + c in physical space

        # Filter to points near expected shock location
        expected_slope = np.tan(np.radians(beta_deg))
        dist_from_expected = np.abs(shock_y - shock_x * expected_slope)
        close_mask = dist_from_expected < 0.05      # within 5 cm of theoretical line
        shock_x = shock_x[close_mask]
        shock_y = shock_y[close_mask]

        if len(shock_x) < 20:
            continue
                
        coeffs = np.polyfit(shock_x, shock_y, 1)
        m = coeffs[0]
        angle_deg = np.degrees(np.arctan(m))
        # Ensure angle is positive and in the correct quadrant (0–90°)
        if angle_deg < 0:
            angle_deg += 180
        if angle_deg > 90:
            angle_deg = 180 - angle_deg
        beta_angles.append(angle_deg)
    return beta_angles


def rankine_hugoniot(mach, beta_deg):
    """Return post-shock P, T, rho and ratios using R-H relations."""
    beta  = np.radians(beta_deg)
    Mn1   = mach * np.sin(beta)

    p_ratio   = (2 * GAMMA * Mn1**2 - (GAMMA - 1)) / (GAMMA + 1)
    rho_ratio = (GAMMA + 1) * Mn1**2 / ((GAMMA - 1) * Mn1**2 + 2)
    T_ratio   = p_ratio / rho_ratio
    Mn2_sq    = ((GAMMA - 1) * Mn1**2 + 2) / (2 * GAMMA * Mn1**2 - (GAMMA - 1))

    P2   = P_INF   * p_ratio
    T2   = T_INF   * T_ratio
    rho2 = RHO_INF * rho_ratio
    a2   = np.sqrt(GAMMA * R_AIR * T2)

    return {
        'pressure': P2, 'temperature': T2, 'density': rho2,
        'p_ratio': p_ratio, 'rho_ratio': rho_ratio, 'T_ratio': T_ratio,
        'Mn1': Mn1, 'Mn2_sq': Mn2_sq, 'a2': a2,
    }


def compute_post_mach(mach, beta_deg, wedge_deg):
    """Compute post-shock Mach number."""
    rh     = rankine_hugoniot(mach, beta_deg)
    Mn2    = np.sqrt(rh['Mn2_sq'])
    beta_w = beta_deg - wedge_deg
    if abs(beta_w) < 0.1:
        return np.nan
    return Mn2 / np.sin(np.radians(beta_w))


def compute_post_velocity(mach, beta_deg, wedge_deg):
    """Compute post-shock velocity magnitude."""
    rh = rankine_hugoniot(mach, beta_deg)
    M2 = compute_post_mach(mach, beta_deg, wedge_deg)
    if M2 is None or np.isnan(M2):
        return np.nan
    return M2 * rh['a2']


def build_features(xf, yf, mach, angle, beta_pred):
    """
    Build the 7 input features for ANN for every fluid grid point.
    Returns a numpy array of shape (n_points, 7) and region_encoded array.
    """
    n = len(xf)
    wedge_slope = np.tan(np.radians(angle))
    beta_r      = np.radians(beta_pred)

    # Features
    x_safe          = np.where(np.abs(xf) < 1e-8, 1e-8, xf)
    y_over_x        = np.clip(yf / x_safe, -10, 10)
    r_from_tip      = np.sqrt(xf**2 + yf**2)
    dist_from_wedge = yf - xf * wedge_slope
    polar_angle     = np.arctan2(yf, xf + 1e-10)
    compressibility = ((GAMMA - 1) / 2 * mach**2)
    mach_squared    = mach**2

    # Assemble input features in the exact order of X_FEATURES
    X = np.column_stack([
        xf, yf,
        np.full(n, mach, dtype=np.float32),
        np.full(n, angle, dtype=np.float32),
        np.full(n, compressibility, dtype=np.float32),
        np.full(n, mach_squared, dtype=np.float32),
        polar_angle,
        r_from_tip,
        dist_from_wedge
    ]).astype(np.float32)

    # region encoding (pre=0, transition=1, post=2)
    dist_from_shock = (-np.sin(beta_r) * xf + np.cos(beta_r) * yf)
    region_encoded = np.where(
        dist_from_shock >  TRANS_HALF, 0,
        np.where(dist_from_shock < -TRANS_HALF, 2, 1)
    ).astype(np.int32)

    return X, region_encoded

def predict_field_ann(mach, angle, grid_res=GRID_RES):
    """
    Predict all 5 flow fields for one (mach, angle) combination.
    Returns:
        XX, YY (meshgrid)
        predictions: dict of target -> 2D array (grid_res x grid_res)
        fluid_mask_2d: 2D boolean array (True where fluid)
        beta_deg: theoretical shock angle (used for region classification)
    """
    beta_deg = compute_shock_angle(mach, angle)
    if np.isnan(beta_deg):
        return None

    x_grid = np.linspace(X_MIN, X_MAX, grid_res)
    y_grid = np.linspace(Y_MIN, Y_MAX, grid_res)
    XX, YY = np.meshgrid(x_grid, y_grid)
    x_flat = XX.ravel().astype(np.float32)
    y_flat = YY.ravel().astype(np.float32)

    # Identify fluid points (exclude wedge interior)
    wedge_slope = np.tan(np.radians(angle))
    fluid_flat = (~((x_flat > 0) & (y_flat < x_flat * wedge_slope))) & (y_flat >= 0)

    xf = x_flat[fluid_flat]
    yf = y_flat[fluid_flat]

    # Build features and scale
    X_raw, region_enc = build_features(xf, yf, mach, angle, beta_deg)

    # Scale each feature group
    coords_scaled   = scaler_coords.transform(X_raw[:, :2])
    mach_scaled     = scaler_mach_in.transform(X_raw[:, 2:3])
    angle_scaled    = scaler_angle.transform(X_raw[:, 3:4])
    compress_scaled = scaler_compress.transform(X_raw[:, 4:5])
    mach_sq_scaled  = scaler_mach_sq.transform(X_raw[:, 5:6])
    polar_scaled    = scaler_polar.transform(X_raw[:, 6:7])
    r_tip_scaled    = scaler_r_tip.transform(X_raw[:, 7:8])
    dist_scaled     = scaler_dist.transform(X_raw[:, 8:9])

    X_scaled = np.hstack([coords_scaled, mach_scaled, angle_scaled,
                          compress_scaled, mach_sq_scaled, polar_scaled,
                          r_tip_scaled, dist_scaled ])

    # Predict all 5 outputs at once
    pred_scaled = model.predict(X_scaled, batch_size=4096, verbose=0)

    # Inverse transform to physical units
    pred_phys = np.hstack([
        scaler_pressure.inverse_transform(pred_scaled[:, 0].reshape(-1, 1)),
        scaler_density.inverse_transform(pred_scaled[:, 1].reshape(-1, 1)),
        scaler_mach_out.inverse_transform(pred_scaled[:, 2].reshape(-1, 1)),
        scaler_velocity.inverse_transform(pred_scaled[:, 3].reshape(-1, 1)),
        scaler_temperature.inverse_transform(pred_scaled[:, 4].reshape(-1, 1))
    ])

    # Assemble full grid arrays (set non-fluid to NaN)
    pred_dict = {}
    for idx, target in enumerate(TARGETS):
        Z_flat = np.full(len(x_flat), np.nan, dtype=np.float32)
        Z_flat[fluid_flat] = pred_phys[:, idx]
        Z = Z_flat.reshape(grid_res, grid_res)
        pred_dict[target] = Z

    #fluid_mask_2d = fluid_flat.reshape(grid_res, grid_res)
    fluid_mask_2d_i = fluid_flat.reshape(grid_res, grid_res)
    sigma = 0.5
    fluid_mask_2d = gaussian_filter(fluid_mask_2d_i, sigma=sigma)

    return XX, YY, pred_dict, fluid_mask_2d, beta_deg

def plot_ansys_style(XX, YY, Z, fluid_mask, mach, angle, beta_deg,
                     target, save_path, rh_theory=None):
    unit = TARGET_UNITS[target]
    cmap = TARGET_CMAPS[target]

    Z_plot = np.where(fluid_mask, Z, np.nan)
    vmin   = np.nanpercentile(Z_plot, 1)
    vmax   = np.nanpercentile(Z_plot, 99)
    levels = np.linspace(vmin, vmax, 21)

    fig, ax = plt.subplots(figsize=(10, 8))
    fig.patch.set_facecolor('white')
    ax.set_facecolor('#E8E8E8')

    cf   = ax.contourf(XX, YY, Z_plot, levels=levels, cmap=cmap, extend='both')
    cbar = plt.colorbar(cf, ax=ax, fraction=0.035, pad=0.02)
    cbar.set_label(f'{target.replace("-", " ").title()} ({unit})', fontsize=11)
    cbar.formatter = matplotlib.ticker.ScalarFormatter(useMathText=True)
    cbar.formatter.set_powerlimits((-2, 4))
    cbar.update_ticks()

    wedge_slope = np.tan(np.radians(angle))
    ax.fill([0, X_MAX, X_MAX, 0], [0, X_MAX * wedge_slope, 0, 0],
            color='#CCCCCC', zorder=5)
    ax.plot([0, X_MAX], [0, X_MAX * wedge_slope], 'k-', lw=1.5, zorder=6)

    x_sh = np.array([0, X_MAX])
    y_sh = x_sh * np.tan(np.radians(beta_deg))
    ax.plot(x_sh[y_sh <= Y_MAX], y_sh[y_sh <= Y_MAX], 'w--', lw=2.0,
            label=f'Theory: β={beta_deg:.1f}°', zorder=7)

    if rh_theory is not None:
        val = rh_theory.get(target)
        if val is not None and not np.isnan(val):
            ax.text(0.98, 0.97, f'R-H: {val:.4g} {unit}',
                    transform=ax.transAxes, ha='right', va='top', fontsize=9,
                    color='white',
                    bbox=dict(boxstyle='round,pad=0.3', facecolor='black', alpha=0.6))

    ax.set_xlim([X_MIN, X_MAX]); ax.set_ylim([Y_MIN, Y_MAX])
    ax.set_xlabel('x (m)', fontsize=11); ax.set_ylabel('y (m)', fontsize=11)
    ax.set_title(
        f'{target.replace("-", " ").title()} — M={mach}, θ={angle}°  (ANN)\n'
        f'Shock angle β={beta_deg:.2f}° (theory)', fontsize=12)
    ax.legend(loc='upper left', fontsize=9, framealpha=0.7)
    ax.set_aspect('equal'); ax.grid(False)

    plt.tight_layout()
    plt.savefig(save_path, dpi=150, bbox_inches='tight', facecolor='white')
    plt.close()

def plot_line_profiles(pred_dict, XX, YY, fluid_mask, mach, angle, beta_deg, save_path):
    y_probe       = 0.15
    x_shock_probe = y_probe / np.tan(np.radians(beta_deg))
    if not (X_MIN < x_shock_probe < X_MAX):
        y_probe       = 0.08
        x_shock_probe = y_probe / np.tan(np.radians(beta_deg))

    fig, axes = plt.subplots(2, 3, figsize=(18, 10))
    fig.suptitle(
        f'Flow Variable Line Profiles — M={mach}, θ={angle}°, β={beta_deg:.1f}° (ANN)\n'
        f'Horizontal probe at y={y_probe:.2f} m  |  '
        f'Shock crosses at x={x_shock_probe:.3f} m',
        fontsize=12)
    axes_flat   = axes.ravel()
    wedge_slope = np.tan(np.radians(angle))

    for i, target in enumerate(TARGETS):
        ax  = axes_flat[i]
        Z   = pred_dict[target]
        Z_plot = np.where(fluid_mask, Z, np.nan)

        x_grid  = XX[0, :]
        y_grid  = YY[:, 0]
        row_idx = np.argmin(np.abs(y_grid - y_probe))
        z_line  = Z_plot[row_idx, :].copy()

        wedge_mask = (x_grid > 0) & (y_probe < x_grid * wedge_slope)
        z_line[wedge_mask | np.isnan(z_line)] = np.nan

        unit = TARGET_UNITS[target]
        ax.plot(x_grid, z_line, 'b-', lw=2, label='ANN prediction')
        ax.axvline(x_shock_probe, color='red', ls='--', lw=1.5,
                   label=f'Shock (x={x_shock_probe:.3f} m)')

        beta_r = np.radians(beta_deg)
        x_tl   = (y_probe - TRANS_HALF * np.cos(beta_r)) / np.tan(beta_r)
        x_tr   = (y_probe + TRANS_HALF * np.cos(beta_r)) / np.tan(beta_r)
        ax.axvspan(x_tl, x_tr, alpha=0.15, color='red', label='Transition band')

        ax.set_xlabel('x (m)', fontsize=9)
        ax.set_ylabel(f'{target} ({unit})', fontsize=9)
        ax.set_title(target, fontsize=10)
        ax.legend(fontsize=7); ax.grid(alpha=0.3)
        ax.set_xlim([X_MIN, X_MAX])

    axes_flat[-1].axis('off')
    plt.tight_layout()
    plt.savefig(save_path, dpi=150, bbox_inches='tight')
    plt.close()

def plot_extrapolation_degradation(parity_data, save_path):
    M_MIN, M_MAX = 2.0, 4.0
    A_MAX        = 20.0
    points       = []

    for target in TARGETS:
        for v in parity_data[target]:
            if np.isnan(v.get('theory', np.nan)) or v['theory'] == 0:
                continue
            m, a  = v['mach'], v['angle']
            m_out = max(0, (M_MIN - m) / M_MIN * 100 if m < M_MIN
                        else (m - M_MAX) / M_MAX * 100 if m > M_MAX else 0)
            a_out = max(0, (a - A_MAX) / A_MAX * 100 if a > A_MAX else 0)
            dist  = max(m_out, a_out)
            err   = abs(v['pred'] - v['theory']) / abs(v['theory']) * 100
            points.append({'dist': dist, 'err': err, 'target': target})

    if not points:
        return

    df  = pd.DataFrame(points)
    fig, axes = plt.subplots(1, 2, figsize=(16, 6))
    fig.suptitle(
        'Prediction Accuracy vs Extrapolation Distance (ANN)', fontsize=13)

    target_colors = {
        'pressure':           '#E84855',
        'temperature':        '#FF9F1C',
        'density':            '#2EC4B6',
        'velocity-magnitude': '#6B4226',
        'mach-number':        '#3A0CA3',
    }
    for target in TARGETS:
        sub = df[df['target'] == target].sort_values('dist')
        axes[0].scatter(sub['dist'], sub['err'],
                        c=target_colors[target], s=60, alpha=0.8, label=target)

    axes[0].axvline(0, color='green', lw=2, label='Training boundary')
    axes[0].set_xlabel('Distance outside training boundary (%)', fontsize=11)
    axes[0].set_ylabel('Post-shock prediction error (%)', fontsize=11)
    axes[0].set_title('Per-target error vs extrapolation distance')
    axes[0].legend(fontsize=8); axes[0].grid(alpha=0.3)

    mean_by_dist = df.groupby('dist')['err'].mean().reset_index()
    axes[1].scatter(mean_by_dist['dist'], mean_by_dist['err'],
                    s=120, c='#2E86AB', zorder=5, edgecolors='white')
    axes[1].axvline(0,   color='green',  lw=2,   label='Training boundary')
    axes[1].axhline(5.0, color='red',    lw=1.5, ls='--', label='5% threshold')
    axes[1].axhline(2.0, color='orange', lw=1.5, ls='--', label='2% threshold')
    if len(mean_by_dist) > 2:
        z    = np.polyfit(mean_by_dist['dist'], mean_by_dist['err'], 2)
        xfit = np.linspace(0, mean_by_dist['dist'].max(), 100)
        axes[1].plot(xfit, np.polyval(z, xfit), 'b--', alpha=0.7, lw=2, label='Trend')
    axes[1].set_xlabel('Distance outside training boundary (%)', fontsize=11)
    axes[1].set_ylabel('Mean error across all 5 targets (%)', fontsize=11)
    axes[1].set_title('Mean error vs extrapolation distance')
    axes[1].legend(fontsize=8); axes[1].grid(alpha=0.3)

    plt.tight_layout()
    plt.savefig(save_path, dpi=150, bbox_inches='tight')
    plt.close()

          f"   {info['n_pre']:8,}  {info['n_trans']:8,}  "
          f"   {info['n_post']:8,}  {total:8,}")

    w("── SECTION 8: INFERENCE TIMING BENCHMARK ─────────────────────────────────")
    w()
    w(f"  Grid resolution:  {GRID_RES} × {GRID_RES} = {GRID_RES**2:,} points")
    w(f"  Fluid points per case: ~230,000–241,000 (wedge excluded)")
    w()
    timing_vals = [info.get('pred_time_s', 0)
                   for info in all_results.values()
                   if info.get('pred_time_s', 0) > 0]

    w(f"  {'Mach':>6}  {'Angle':>7}  {'Cat':>4}  {'Time(s)':>9}")

    for (mach, angle, cat), info in all_results.items():

    w("=" * 78)

    print(f"  ╚══════════════════════════════════════════════════════╝")

def main():
    print("=" * 70)
    print("  ANN FLOW FIELD PREDICTION  (Oblique Shock ML)")
    print("=" * 70)
    print_mem("start")

    all_results = {}
    parity_data = {t: [] for t in TARGETS}

    for case_idx, (mach, angle, cat) in enumerate(ALL_CASES):
        print(f"\n{'─' * 60}")
        print(f"  Case {case_idx + 1}/{len(ALL_CASES)} [{cat}]  "
              f"M={mach}, θ={angle}°")

        beta_deg = compute_shock_angle(mach, angle)
        if np.isnan(beta_deg):
            print(f"  ⚠ Detached shock — skipping case")
            continue
        print(f"  β = {beta_deg:.3f}°")

        rh  = rankine_hugoniot(mach, beta_deg)
        M2  = compute_post_mach(mach, beta_deg, angle)
        V2  = compute_post_velocity(mach, beta_deg, angle)
        rh_theory = {
            'pressure':           rh['pressure'],
            'temperature':        rh['temperature'],
            'density':            rh['density'],
            'velocity-magnitude': V2 if (V2 is not None and not np.isnan(V2))
                                  else np.nan,
            'mach-number':        M2 if (M2 is not None and not np.isnan(M2))
                                  else np.nan,
        }
        print(f"  R-H: P={rh['pressure']:.0f} Pa  "
              f"T={rh['temperature']:.1f} K  ρ={rh['density']:.4f} kg/m³")

        case_dir = os.path.join(FIGURES_DIR,
                                f"M{mach}_angle{angle}_{cat}")
        os.makedirs(case_dir, exist_ok=True)

        t_start = time.time()

        # Predict fields
        res = predict_field_ann(mach, angle)
        if res is None:
            print("  ✗ Prediction failed (no solution)")
            continue
        XX, YY, pred_dict, fluid_mask, beta_deg = res
        beta_angles = beta_angle(XX, YY, pred_dict, fluid_mask, beta_deg)

        if beta_angles:
            beta_pred = np.mean(beta_angles)
        else:
            beta_pred = None
        if beta_pred is not None and beta_deg != 0:
            beta_error = abs((beta_pred - beta_deg) / beta_deg * 100)
        else:
            beta_error = np.nan
        
        # Compute region counts and collect post-shock/transition means
        x_flat = XX.ravel()
        y_flat = YY.ravel()
        wedge_slope = np.tan(np.radians(angle))
        fluid_flat = (~((x_flat > 0) & (y_flat < x_flat * wedge_slope))) & (y_flat >= 0)
        xf = x_flat[fluid_flat]
        yf = y_flat[fluid_flat]
        region_beta = beta_pred if beta_pred is not None else beta_deg
        _, region_enc = build_features(xf, yf, mach, angle, region_beta)

        case_region_counts = {
            'n_pre':   int((region_enc == 0).sum()),
            'n_trans': int((region_enc == 1).sum()),
            'n_post':  int((region_enc == 2).sum()),
        }

        # Save individual contours and collect parity data
        case_errors = {}
        for target in TARGETS:
            Z = pred_dict[target]
            plot_ansys_style(XX, YY, Z, fluid_mask, mach, angle, beta_deg,
                             target, os.path.join(case_dir, f"{target}_contour.png"),
                             rh_theory)

            # Post-shock mean
            post_mask = (region_enc == 2)
            trans_mask = (region_enc == 1)
            Z_fluid = Z.ravel()[fluid_flat]
            post_mean = np.nanmean(Z_fluid[post_mask]) if np.any(post_mask) else np.nan
            trans_mean = np.nanmean(Z_fluid[trans_mask]) if np.any(trans_mask) else np.nan

            parity_data[target].append({
                'mach':       mach,
                'angle':      angle,
                'cat':        cat,
                'pred':       post_mean,
                'pred_trans': trans_mean,
                'theory':     rh_theory.get(target, np.nan),
            })

            theory_val = rh_theory.get(target, np.nan)
            if np.isfinite(theory_val) and theory_val != 0:
                case_errors[target] = abs(post_mean - theory_val) / abs(theory_val) * 100

        # All-targets panel
        plot_all_targets_panel(pred_dict, XX, YY, fluid_mask, mach, angle, beta_deg,
                               os.path.join(case_dir, "all_targets_panel.png"),
                               rh_theory, case_errors)
        print(f"  ✓ Panel plot saved")

        # Line profiles
        plot_line_profiles(pred_dict, XX, YY, fluid_mask, mach, angle, beta_deg,
                           os.path.join(case_dir, "line_profiles.png"))
        print(f"  ✓ Line profiles saved")

        pred_time = time.time() - t_start
        all_results[(mach, angle, cat)] = {
            'beta':        beta_deg,
            'beta_pred':   beta_pred,
            'beta_error':  beta_error,
            'pred_time_s': round(pred_time, 1),
            **case_region_counts,
        }

        gc.collect()
        print_mem(f"after M={mach}, θ={angle}")

    # Summary plots
    print("\n  Generating summary plots...")
    plot_parity_all_targets(parity_data,
                            os.path.join(FIGURES_DIR, "parity_all_targets_ann.png"))
    print("  ✓ Parity plot saved")

    for target in TARGETS:
        plot_error_heatmap(parity_data, target,
                           os.path.join(FIGURES_DIR, f"error_heatmap_{target}_ann.png"))
    print("  ✓ Error heatmaps saved")

    plot_prediction_coverage(parity_data, ALL_CASES,
                             os.path.join(FIGURES_DIR, "prediction_coverage_ann.png"))
    print("  ✓ Coverage map saved")

    plot_extrapolation_degradation(parity_data,
                                   os.path.join(FIGURES_DIR, "extrapolation_degradation_ann.png"))
    print("  ✓ Extrapolation degradation plot saved")

    plot_error_vs_input(parity_data, ALL_CASES,
                        os.path.join(FIGURES_DIR, "error_vs_input_ann.png"))
    print("  ✓ Error vs input plot saved")

    plot_region_error_comparison(parity_data,
                                 os.path.join(FIGURES_DIR, "region_error_comparison_ann.png"))
    print("  ✓ Region error comparison saved")

    # CSV summary
    rows = []
    for target in TARGETS:
        for v in parity_data[target]:
            err = (abs(v['pred'] - v['theory']) / abs(v['theory']) * 100
                   if (not np.isnan(v.get('theory', np.nan))
                       and v['theory'] != 0) else np.nan)
            rows.append({
                'target':    target,
                'mach':      v['mach'],
                'angle':     v['angle'],
                'category':  v['cat'],
                'predicted': v['pred'],
                'pred_trans':v.get('pred_trans', np.nan),
                'theory_rh': v['theory'],
                'error_pct': err,
            })
    pd.DataFrame(rows).to_csv(os.path.join(OUTPUT_DIR, "predictions_summary_ann.csv"),
                              index=False)
    print("  ✓ predictions_summary_ann.csv saved")

    # Blackbox report
    generate_blackbox_report(all_results, parity_data, OUTPUT_DIR)

    cleanup_and_compress(keep_summary=True)

    print_mem("final")
    print("\n" + "=" * 70)
    print("  ANN PREDICTION COMPLETE")
    print(f"  Figures: {FIGURES_DIR}")
    print(f"  Figures: {FIGURES_DIR} (compressed to {OUTPUT_DIR}/figures_compressed.zip)")
    print(f"  Report:  {OUTPUT_DIR}/blackbox_ann_report.txt")
    print("=" * 70)


if __name__ == "__main__":
    main()
