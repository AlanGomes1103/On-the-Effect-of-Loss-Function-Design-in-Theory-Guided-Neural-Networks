# Combined extrapolation-degradation overlay across all 8 model variants.

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

# ===============================
# GPU OPTIMISATION
# ===============================

gpus = tf.config.list_physical_devices('GPU')
if gpus:
    for gpu in gpus:
        tf.config.experimental.set_memory_growth(gpu, True)
    print(f'✓ GPU memory growth enabled for {len(gpus)} GPU(s)')

# ===============================
# LOAD CSV FILES
# ===============================

PRED_FILES = {
    "Residual bMSE"        : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_bMSE_res.csv",
    "Residual wMSE"        : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_wMSE_res.csv",
    "Residual RHLoss"      : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_RHloss_res.csv",
    "Residual wRHLoss"     : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_wRHloss_res.csv",
    "Sequential bMSE"      : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_bMSE_seq.csv",
    "Sequential wMSE"      : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_wMSE_seq.csv",
    "Sequential RHLoss"    : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_RHloss_seq.csv",
    "Sequential wRHLoss"   : "/kaggle/input/datasets/general005/pred-summary/predictions_summary_ann_wRHloss_seq.csv",
}

# Training envelope bounds — must match your training/prediction scripts
M_MIN, M_MAX = 2.0, 4.0
A_MAX        = 20.0

# Visual encoding: colour = architecture, marker/linestyle = loss function
ARCH_COLOR = {"Residual": "#2E86AB", "Sequential": "#E84855"}
LOSS_STYLE = {
    "bMSE":    {"marker": "o", "linestyle": "-"},
    "wMSE":    {"marker": "s", "linestyle": "--"},
    "RHLoss":  {"marker": "^", "linestyle": "-."},
    "wRHLoss": {"marker": "x", "linestyle": ":"},
}


def compute_distance(mach, angle):
    """Normalized L-inf distance outside the training envelope (%).
    Matches plot_extrapolation_degradation() in the prediction script.
    Rounded to avoid float-precision duplicate bins when grouping."""
    if mach < M_MIN:
        m_out = (M_MIN - mach) / M_MIN * 100
    elif mach > M_MAX:
        m_out = (mach - M_MAX) / M_MAX * 100
    else:
        m_out = 0.0
    a_out = max(0.0, (angle - A_MAX) / A_MAX * 100)
    return round(max(m_out, a_out), 4)

# ===============================
# PLOT FIGURE
# ===============================

fig, ax = plt.subplots(figsize=(11, 7))

for label, path in PRED_FILES.items():
    try:
        df = pd.read_csv(path)
    except FileNotFoundError:
        print(f"  ⚠ Skipping '{label}': file not found at {path}")
        continue

    df = df.dropna(subset=["error_pct"])          # drop cases with no theory value
    df["dist"] = df.apply(lambda r: compute_distance(r["mach"], r["angle"]), axis=1)

    agg = (df.groupby("dist")["error_pct"]
             .mean()
             .reset_index()
             .sort_values("dist"))

    arch  = "Residual" if "Residual" in label else "Sequential"
    loss  = label.split()[-1]
    style = LOSS_STYLE[loss]

    ax.plot(agg["dist"], agg["error_pct"],
            color=ARCH_COLOR[arch], marker=style["marker"],
            linestyle=style["linestyle"], linewidth=2, markersize=7,
            label=label, alpha=0.9)


ax.axvline(0, color="green", lw=2, label="Training boundary")
ax.axhline(5.0, color="red", lw=1.2, ls="--", alpha=0.6, label="5% threshold")
ax.axhline(2.0, color="orange", lw=1.2, ls="--", alpha=0.6, label="2% threshold")

ax.set_xlabel("Distance outside training boundary (%)", fontsize=12)
ax.set_ylabel("Mean post-shock error across all 5 targets (%)", fontsize=12)
ax.set_title("Extrapolation Degradation — All 8 Model Variants", fontsize=14)
ax.grid(alpha=0.3)
ax.legend(fontsize=9, ncol=2, loc="upper left")

plt.tight_layout()
plt.savefig("extrapolation_degradation_overlay.png", dpi=300, bbox_inches="tight")
plt.close()
print("Saved: extrapolation_degradation_overlay.png")
