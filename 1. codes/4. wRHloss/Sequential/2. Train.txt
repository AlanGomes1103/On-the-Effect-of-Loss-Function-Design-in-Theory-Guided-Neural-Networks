def rh_compliance_score(y_pred_phys, y_theory_phys, post_mask):

    var_names = ['Pressure', 'Density', 'Mach', 'Velocity', 'Temperature']
    
    pred_post   = y_pred_phys[post_mask]
    theory_post = y_theory_phys[post_mask]
    
    denom  = np.where(np.abs(theory_post) > 1e-8, np.abs(theory_post), 1e-8)
    errors = np.abs(pred_post - theory_post) / denom * 100  # shape (N_post, 5)
    
    per_var = {var: float(errors[:, i].mean()) for i, var in enumerate(var_names)}
    overall_score = float(errors.mean())
    
    return overall_score, per_var

def comprehensive_diagnostics(model, X_train, y_train, X_val, y_val, history,
                                var_names=None,
                                val_region_mask=None,
                                val_pre_shock_mask=None,
                                val_post_shock_mask=None,
                                output_scalers=None,    
                                y_val_raw=None,         
                                y_train_raw=None,
                                y_theory_phys=None,
                                post_mask_bool=None):    

    if var_names is None:
        var_names = ['Pressure', 'Density', 'Mach', 'Velocity', 'Temperature']

    print("\n" + "=" * 70)
    print("COMPREHENSIVE FIT DIAGNOSTICS")
    print("=" * 70)

    train_pred = model.predict(X_train, verbose=0, batch_size=BATCH_SIZE)
    val_pred   = model.predict(X_val,   verbose=0, batch_size=BATCH_SIZE)

    if output_scalers is not None:
        train_pred_phys = np.hstack([
            sc.inverse_transform(train_pred[:, i].reshape(-1, 1))
            for i, sc in enumerate(output_scalers)
        ])
        val_pred_phys = np.hstack([
            sc.inverse_transform(val_pred[:, i].reshape(-1, 1))
            for i, sc in enumerate(output_scalers)
        ])
        y_train_phys = y_train_raw if y_train_raw is not None else y_train
        y_val_phys   = y_val_raw   if y_val_raw   is not None else y_val
    else:
        # Fallback: use scaled arrays (metrics will be in scaled units)
        train_pred_phys = train_pred
        val_pred_phys   = val_pred
        y_train_phys    = y_train
        y_val_phys      = y_val
        
    # ------------------------------------------------------------------
    # 1. OVERALL METRICS
    # ------------------------------------------------------------------
    print("\n1. OVERALL METRICS COMPARISON")
    print("-" * 70)

    train_r2_overall   = r2_score(y_train, train_pred)
    val_r2_overall     = r2_score(y_val,   val_pred)
    train_mae_overall  = mean_absolute_error(y_train_phys, train_pred_phys)
    val_mae_overall    = mean_absolute_error(y_val_phys,   val_pred_phys)
    train_mse_overall  = mean_squared_error(y_train_phys,  train_pred_phys)
    val_mse_overall    = mean_squared_error(y_val_phys,    val_pred_phys)
    train_rmse_overall = np.sqrt(train_mse_overall)
    val_rmse_overall   = np.sqrt(val_mse_overall)
    denom_train        = np.abs(y_train_phys)
    denom_val          = np.abs(y_val_phys)
    train_mape_overall = np.mean(np.abs(y_train_phys - train_pred_phys) /
                                 np.where(denom_train > 1e-8, denom_train, 1e-8)) * 100
    val_mape_overall   = np.mean(np.abs(y_val_phys   - val_pred_phys)   /
                                 np.where(denom_val   > 1e-8, denom_val,   1e-8)) * 100

    print(f"{'Metric':<20} {'Training':<15} {'Validation':<15} {'Difference':<15}")
    print("-" * 70)
    print(f"{'R²':<20} {train_r2_overall:<15.6f} {val_r2_overall:<15.6f} "
          f"{abs(train_r2_overall - val_r2_overall):<15.6f}")
    print(f"{'MAE':<20} {train_mae_overall:<15.6f} {val_mae_overall:<15.6f} "
          f"{abs(train_mae_overall - val_mae_overall):<15.6f}")
    print(f"{'MSE':<20} {train_mse_overall:<15.6f} {val_mse_overall:<15.6f} "
          f"{abs(train_mse_overall - val_mse_overall):<15.6f}")
    print(f"{'RMSE':<20} {train_rmse_overall:<15.6f} {val_rmse_overall:<15.6f} "
          f"{abs(train_rmse_overall - val_rmse_overall):<15.6f}")
    print(f"{'MAPE':<20} {train_mape_overall:<15.6f} {val_mape_overall:<15.6f} "
          f"{abs(train_mape_overall - val_mape_overall):<15.6f}")

    r2_drop      = (1 - val_r2_overall / train_r2_overall) * 100 if train_r2_overall > 0 else 0
    mae_increase = (val_mae_overall / train_mae_overall - 1) * 100 if train_mae_overall > 0 else 0
    print(f"\nR² drop:        {r2_drop:.2f}%")
    print(f"MAE increase:   {mae_increase:.2f}%")

    # ------------------------------------------------------------------
    # 2. TRANSITION REGION METRICS
    # ------------------------------------------------------------------
    r2_trans       = None
    mae_trans      = None
    mse_trans      = None
    rmse_trans     = None
    mape_trans     = None
    y_val_trans    = None
    pred_val_trans = None
    r2_pre         = None
    mae_pre        = None

    if val_region_mask is not None:
        if len(val_region_mask) != len(y_val):
            print("⚠️  val_region_mask length mismatch — skipping region metrics.")
        else:
            y_val_trans    = y_val_phys[val_region_mask]
            pred_val_trans = val_pred_phys[val_region_mask]

            if len(y_val_trans) == 0:
                print("⚠️  No transition points in validation set.")
            else:
                print("\n" + "=" * 70)
                print("TRANSITION REGION (SHOCK) METRICS")
                print("=" * 70)
                r2_trans  = r2_score(y_val_trans, pred_val_trans)
                mae_trans = mean_absolute_error(y_val_trans, pred_val_trans)
                print(f"R²  (transition): {r2_trans:.6f}")
                print(f"MAE (transition): {mae_trans:.6f}")

                mse_trans  = mean_squared_error(y_val_trans, pred_val_trans)
                rmse_trans = np.sqrt(mse_trans)
                denom_trans = np.abs(y_val_trans)
                mape_trans = np.mean(
                    np.abs(y_val_trans - pred_val_trans) /
                    np.where(denom_trans > 1e-8, denom_trans, 1e-8)
                ) * 100
                print(f"MSE (transition): {mse_trans:.6f}")
                print(f"RMSE(transition): {rmse_trans:.6f}")
                print(f"MAPE(transition): {mape_trans:.4f}%")

                print("\nPer-variable transition metrics:")
                print(f"{'Variable':<15} {'R²':<12} {'MAE':<12} {'MSE':<14} {'RMSE':<12} {'MAPE %':<12}")
                print("-" * 75)
                for i, var in enumerate(var_names):
                    if i < y_val_trans.shape[1]:
                        rv   = r2_score(y_val_trans[:, i], pred_val_trans[:, i])
                        mav  = mean_absolute_error(y_val_trans[:, i], pred_val_trans[:, i])
                        msev = mean_squared_error(y_val_trans[:, i], pred_val_trans[:, i])
                        rmsv = np.sqrt(msev)
                        dv   = np.abs(y_val_trans[:, i])
                        mapv = np.mean(
                            np.abs(y_val_trans[:, i] - pred_val_trans[:, i]) /
                            np.where(dv > 1e-8, dv, 1e-8)
                        ) * 100
                        print(f"{var:<15} {rv:<12.6f} {mav:<12.4g} {msev:<14.4g} {rmsv:<12.4g} {mapv:<12.4f}")

    # ------------------------------------------------------------------
    # 2b. PRE-SHOCK REGION METRICS
    # ------------------------------------------------------------------
    # FIX 1: uses the passed-in mask parameter, not a global variable
    if val_pre_shock_mask is not None and val_pre_shock_mask.sum() > 0:
        y_val_pre    = y_val_phys[val_pre_shock_mask]
        pred_val_pre = val_pred_phys[val_pre_shock_mask]

        print("\n" + "=" * 70)
        print("PRE-SHOCK (FREESTREAM) REGION METRICS")
        print("=" * 70)
        r2_pre  = r2_score(y_val_pre, pred_val_pre)
        mae_pre = mean_absolute_error(y_val_pre, pred_val_pre)
        print(f"R²  (pre-shock): {r2_pre:.6f}")
        print(f"MAE (pre-shock): {mae_pre:.6f}")

        print("\nPer-variable pre-shock metrics:")
        print(f"{'Variable':<15} {'R²':<12} {'MAE':<12} {'MAPE %':<12}")
        print("-" * 50)
        for i, var in enumerate(var_names):
            if i < y_val_pre.shape[1]:
                rv    = r2_score(y_val_pre[:, i], pred_val_pre[:, i])
                mav   = mean_absolute_error(y_val_pre[:, i], pred_val_pre[:, i])
                denom = np.abs(y_val_pre[:, i])
                mape  = np.mean(mav / np.where(denom > 1e-8, denom, 1e-8)) * 100
                print(f"{var:<15} {rv:<12.6f} {mav:<12.4g} {mape:<12.4f}")

    # ------------------------------------------------------------------
    # 2c. POST-SHOCK REGION METRICS
    # ------------------------------------------------------------------
    if val_post_shock_mask is not None and val_post_shock_mask.sum() > 0:
        y_val_post    = y_val_phys[val_post_shock_mask]
        pred_val_post = val_pred_phys[val_post_shock_mask]

        print("\n" + "=" * 70)
        print("POST-SHOCK (DOWNSTREAM) REGION METRICS")
        print("=" * 70)
        r2_post  = r2_score(y_val_post, pred_val_post)
        mae_post = mean_absolute_error(y_val_post, pred_val_post)
        print(f"R²  (post-shock): {r2_pre:.6f}")
        print(f"MAE (post-shock): {mae_pre:.6f}")

        print("\nPer-variable post-shock metrics:")
        print(f"{'Variable':<15} {'R²':<12} {'MAE':<12} {'MAPE %':<12}")
        print("-" * 50)
        for i, var in enumerate(var_names):
            if i < y_val_post.shape[1]:
                rv    = r2_score(y_val_post[:, i], pred_val_post[:, i])
                mav   = mean_absolute_error(y_val_post[:, i], pred_val_post[:, i])
                denom = np.abs(y_val_post[:, i])
                mape  = np.mean(mav / np.where(denom > 1e-8, denom, 1e-8)) * 100
                print(f"{var:<15} {rv:<12.6f} {mav:<12.4g} {mape:<12.4f}")

    # ------------------------------------------------------------------
    # 2d. RANKINE-HUGONIOT COMPLIANCE SCORE
    # ------------------------------------------------------------------
    rh_score        = None
    rh_per_var      = None

    if y_theory_phys is not None and post_mask_bool is not None:
        rh_score, rh_per_var = rh_compliance_score(
            val_pred_phys, y_theory_phys, post_mask_bool
        )
        print("\n" + "=" * 70)
        print("RANKINE-HUGONIOT COMPLIANCE (Post-Shock Points)")
        print("=" * 70)
        print(f"Overall R-H compliance error: {rh_score:.4f}%")
        print(f"\n{'Variable':<15} {'R-H MAPE %':<12}")
        print("-" * 30)
        for var, score in rh_per_var.items():
            print(f"{var:<15} {score:<12.4f}")

    # ------------------------------------------------------------------
    # 3. PER-VARIABLE R²
    # ------------------------------------------------------------------
    print("\n2. PER-VARIABLE ANALYSIS")
    print("-" * 105)

    var_issues = []
    print(f"{'Variable':<15} {'Train R²':<12} {'Val R²':<12} {'Drop %':<10} "
          f"{'Val MAE':<12} {'Val MSE':<14} {'Val RMSE':<12} {'Val MAPE%':<12} {'Status'}")
    print("-" * 105)
    for i, var in enumerate(var_names):
        if i < y_train.shape[1]:
            tr   = r2_score(y_train[:, i], train_pred[:, i])
            vl   = r2_score(y_val_phys[:, i],   val_pred_phys[:, i])
            dp   = (1 - vl / tr) * 100 if tr > 0 else 0
            vmae = mean_absolute_error(y_val_phys[:, i], val_pred_phys[:, i])
            vmse = mean_squared_error(y_val_phys[:, i],  val_pred_phys[:, i])
            vrms = np.sqrt(vmse)
            dv   = np.abs(y_val_phys[:, i])
            vmap = np.mean(
                np.abs(y_val_phys[:, i] - val_pred_phys[:, i]) /
                np.where(dv > 1e-8, dv, 1e-8)
            ) * 100

            if tr < 0.85:
                status = "⚠️ Underfit"; var_issues.append((var, "underfit"))
            elif dp > 5:
                status = "⚠️ Overfit";  var_issues.append((var, "overfit"))
            else:
                status = "✅ Good"
            print(f"{var:<15} {tr:<12.6f} {vl:<12.6f} {dp:<10.2f} "
                  f"{vmae:<12.4g} {vmse:<14.4g} {vrms:<12.4g} {vmap:<12.4f} {status}")

    # ------------------------------------------------------------------
    # 4. LOSS CURVE ANALYSIS
    # ------------------------------------------------------------------
    print("\n3. LOSS CURVE ANALYSIS")
    print("-" * 70)

    train_loss_history = history.history['loss']
    val_loss_history   = history.history['val_loss']

    final_train_loss = train_loss_history[-1]
    final_val_loss   = val_loss_history[-1]
    min_val_loss     = min(val_loss_history)
    best_epoch       = np.argmin(val_loss_history) + 1

    # FIX 3: use min_val_loss vs final_train_loss to avoid Dropout-inflated gap
    gap         = min_val_loss - final_train_loss
    gap_percent = abs(gap / final_train_loss) * 100 if final_train_loss > 0 else 0

    print(f"Final Training Loss:   {final_train_loss:.6f}")
    print(f"Final Validation Loss: {final_val_loss:.6f}")
    print(f"Best Validation Loss:  {min_val_loss:.6f} (epoch {best_epoch})")
    print(f"Loss Gap (best val vs final train): {gap:.6f} ({gap_percent:.1f}%)")

    if len(val_loss_history) >= 20:
        recent = val_loss_history[-20:]
        if np.mean(recent[-10:]) > np.mean(recent[:10]) * 1.05:
            print("\n⚠️  WARNING: Validation loss rising in recent epochs — possible overfitting.")

    if len(train_loss_history) >= 10:
        recent_tr = train_loss_history[-10:]
        if np.std(recent_tr) / np.mean(recent_tr) < 0.01:
            print("\n✓ Training loss has converged.")
        else:
            print("\n⚠️  Training loss still changing — may benefit from more epochs.")

    # ------------------------------------------------------------------
    # 5. PRIMARY DIAGNOSIS
    # ------------------------------------------------------------------
    print("\n" + "=" * 70)
    print("PRIMARY DIAGNOSIS")
    print("=" * 70)

    is_underfitting = train_r2_overall < 0.85
    # FIX 3: overfitting check uses min_val_loss direction, not raw negative gap
    is_overfitting  = (r2_drop > 5) or (gap_percent > 50 and min_val_loss > final_train_loss)
    is_good         = not is_underfitting and not is_overfitting

    if is_underfitting:
        print("\n❌ PRIMARY ISSUE: UNDERFITTING")
        print(f"  • Training R² = {train_r2_overall:.3f} (target: >0.90)")
        print(f"  • Validation R² = {val_r2_overall:.3f}")
        print("\n📋 RECOMMENDED ACTIONS:")
        print("  1. Increase model capacity (add neurons / layers)")
        print("  2. Reduce dropout and L2 regularisation")
        print("  3. Train for more epochs")
        print("  4. Review feature engineering")

    elif is_overfitting:
        print("\n❌ PRIMARY ISSUE: OVERFITTING")
        print(f"  • R² drop = {r2_drop:.1f}% (target: <5%)")
        print(f"  • Loss gap = {gap_percent:.1f}% (target: <20%)")
        print("\n📋 RECOMMENDED ACTIONS:")
        print("  1. Increase dropout and L2 regularisation")
        print("  2. Reduce model complexity")
        print("  3. Stricter early stopping patience")
        print("  4. Get more training data")

    else:
        print("\n✅ MODEL IS WELL-FITTED!")
        print(f"  • Training R² = {train_r2_overall:.3f}")
        print(f"  • Validation R² = {val_r2_overall:.3f}")
        print(f"  • R² drop = {r2_drop:.1f}% ✓")
        print(f"  • Loss gap = {gap_percent:.1f}% ✓")
        print("\n📋 NEXT STEPS:")
        print("  1. Test on held-out test set to confirm")
        print("  2. Analyse per-variable performance")
        print("  3. Review R-H compliance history plot")

    # ------------------------------------------------------------------
    # 6. VARIABLE-SPECIFIC ISSUES
    # ------------------------------------------------------------------
    if var_issues:
        print("\n" + "=" * 70)
        print("VARIABLE-SPECIFIC ISSUES")
        print("=" * 70)
        for var, issue in var_issues:
            if issue == "underfit":
                print(f"\n⚠️  {var}: Underfitting — consider feature engineering for this target.")
            elif issue == "overfit":
                print(f"\n⚠️  {var}: Overfitting — consider stronger regularisation for this target.")

    # ------------------------------------------------------------------
    # 7. HEALTH SCORE
    # ------------------------------------------------------------------
    print("\n" + "=" * 70)
    print("SUMMARY SCORES")
    print("=" * 70)

    health_score = 100
    if train_r2_overall < 0.90:
        health_score -= (0.90 - train_r2_overall) * 200
    if r2_drop > 2:
        health_score -= min(r2_drop * 2, 30)
    if gap_percent > 20:
        health_score -= min((gap_percent - 20) * 0.5, 20)
    health_score = max(0, min(100, health_score))

    print(f"\nModel Health Score: {health_score:.0f}/100")
    if health_score >= 90:   print("Grade: A+ (Excellent)")
    elif health_score >= 80: print("Grade: A  (Very Good)")
    elif health_score >= 70: print("Grade: B  (Good)")
    elif health_score >= 60: print("Grade: C  (Acceptable)")
    else:                    print("Grade: D  (Needs Improvement)")

    print("\n" + "=" * 70)

    # ------------------------------------------------------------------
    # 8. DIAGNOSTIC PLOTS
    # ------------------------------------------------------------------
    print("\n📊 Generating diagnostic plots...")

    fig = plt.figure(figsize=(16, 10))

    ax1 = plt.subplot(2, 3, 1)
    ax1.plot(train_loss_history, label='Training Loss',   linewidth=2)
    ax1.plot(val_loss_history,   label='Validation Loss', linewidth=2)
    ax1.axvline(x=best_epoch - 1, color='red', linestyle='--',
                alpha=0.5, label=f'Best Val (epoch {best_epoch})')
    ax1.set_xlabel('Epoch'); ax1.set_ylabel('Loss')
    ax1.set_title('Loss Curves'); ax1.legend(); ax1.grid(True, alpha=0.3)

    ax2 = plt.subplot(2, 3, 2)
    x_pos        = np.arange(len(var_names[:y_train.shape[1]]))
    train_r2_lst = [r2_score(y_train[:, i], train_pred[:, i]) for i in range(y_train.shape[1])]
    val_r2_lst   = [r2_score(y_val[:, i],   val_pred[:, i])   for i in range(y_val.shape[1])]
    ax2.bar(x_pos - 0.2, train_r2_lst, 0.4, label='Training',   alpha=0.8)
    ax2.bar(x_pos + 0.2, val_r2_lst,   0.4, label='Validation', alpha=0.8)
    ax2.set_xticks(x_pos); ax2.set_xticklabels(var_names[:y_train.shape[1]], rotation=45)
    ax2.set_ylabel('R² Score'); ax2.set_title('R² by Variable')
    ax2.legend(); ax2.grid(True, alpha=0.3)
    ax2.axhline(y=0.95, color='green', linestyle='--', alpha=0.5)

    ax3 = plt.subplot(2, 3, 3)
    loss_gap_curve = np.array(val_loss_history) - np.array(train_loss_history)
    ax3.plot(loss_gap_curve, linewidth=2, color='red')
    ax3.axhline(y=0, color='black', linestyle='-', alpha=0.3)
    ax3.set_xlabel('Epoch'); ax3.set_ylabel('Val Loss − Train Loss')
    ax3.set_title('Loss Gap Evolution'); ax3.grid(True, alpha=0.3)

    ax4 = plt.subplot(2, 3, 4)

def theta_beta_mach(mach, wedge_deg):   
    try:
        mu = np.degrees(np.arcsin(1.0 / mach))
    except (ValueError, ZeroDivisionError):
        return np.nan

    def residual(b):
        br = np.radians(b)
        return (np.tan(np.radians(wedge_deg)) -
                2.0 / np.tan(br) *
                (mach**2 * np.sin(br)**2 - 1.0) /
                (mach**2 * (gamma + np.cos(2 * br)) + 2.0))

    bv = np.linspace(mu + 0.5, 89.0, 800)
    sv = [residual(b) for b in bv]
    crossings = [(bv[i], bv[i+1]) for i in range(len(sv) - 1)
                 if sv[i] * sv[i+1] < 0]
    if not crossings:
        return np.nan
    try:
        return brentq(residual, crossings[0][0], crossings[0][1])
    except Exception:
        return np.nan

def wRHloss(y_true, y_pred):
    y_true_data = y_true[:, :5]
    y_true_theo = y_true[:, 5:10]
    mask        = y_true[:, 10:11]
    weights = tf.constant([
        LOSS_WEIGHTS['pressure'],
        LOSS_WEIGHTS['density'],
        LOSS_WEIGHTS['mach-number'],
        LOSS_WEIGHTS['velocity-magnitude'],
        LOSS_WEIGHTS['temperature'],
    ], dtype=tf.float32)
    sq_errors_data = tf.square(y_true_data - y_pred)
    data_loss = tf.reduce_mean(sq_errors_data * weights)

    sq_errors_phys = tf.square(y_pred - y_true_theo)
    n_post    = tf.maximum(tf.reduce_sum(mask), 1.0)
    phys_loss = tf.reduce_sum(sq_errors_phys * mask) / n_post

    return data_loss + phys_loss

        ax.grid(True, alpha=0.3)

        stats_text = (
            f"R²   = {r2:.6f}\n"
            f"MAE  = {mae:.4g} {unit}\n"
            f"RMSE = {rmse:.4g} {unit}"
        )

                          alpha=0.80, edgecolor='gray', linewidth=0.8))
        ax.legend(fontsize=8, loc='lower right')

    plt.savefig(var["file"], dpi=300, bbox_inches='tight')

from sklearn.metrics import mean_absolute_error as _mae

baseline_mae = _mae(y_val, model.predict(X_val, verbose=0, batch_size=4096))
n_repeats    = 5
rng          = np.random.default_rng(42)
importance   = np.zeros((X_val.shape[1], n_repeats))

for feat_idx in range(X_val.shape[1]):
    for rep in range(n_repeats):
        X_permuted               = X_val.copy()
        X_permuted[:, feat_idx]  = rng.permutation(X_permuted[:, feat_idx])
        perm_pred                = model.predict(X_permuted, verbose=0, batch_size=4096)
        importance[feat_idx, rep] = _mae(y_val, perm_pred) - baseline_mae

imp_mean = importance.mean(axis=1)
imp_std  = importance.std(axis=1)
order    = np.argsort(imp_mean)[::-1]

print(f"\n  Baseline MAE: {baseline_mae:.6f}")
print(f"\n  {'Feature':<28} {'Importance (ΔMAE)':<20} {'±Std':<12} {'%':<8}")
print("  " + "-" * 68)
total_imp = imp_mean.sum() if imp_mean.sum() > 0 else 1.0
for rank, idx in enumerate(order):
    pct = imp_mean[idx] / total_imp * 100
    print(f"  {rank+1:>2}. {X_FEATURES[idx]:<24} "
          f"{imp_mean[idx]:<20.6f} "
          f"±{imp_std[idx]:<10.6f} "
          f"{pct:<8.1f}%")

fig_imp, ax_imp = plt.subplots(figsize=(10, 5))

            imp_mean[order[::-1]],
